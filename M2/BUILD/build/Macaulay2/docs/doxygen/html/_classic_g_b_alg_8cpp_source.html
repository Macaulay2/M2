117ClassicGBAlg::ClassicGBAlg(
118 const Basis& basis,
119 Reducer& reducer,
120 int monoLookupType,
121 bool preferSparseReducers,
122 size_t queueType
123):
124 mCallback(nullptr),
125 mBreakAfter(0),
126 mPrintInterval(0),
127 mSPairGroupSize(reducer.preferredSetSize()),
128 mUseAutoTopReduction(true),
129 mUseAutoTailReduction(false),
130 mRing(*basis.getPolyRing()),
131 mReducer(reducer),
132 mBasis(mRing,
133 MonoLookup::makeFactory(
134 basis.getPolyRing()->monoid(),
135 monoLookupType
136 )->make(preferSparseReducers, true)
137 ),
138 mSPairs(mBasis, preferSparseReducers),
139 mSPolyReductionCount(0)
140{
141 (void)queueType;
142 // Reduce and insert the generators of the ideal into the starting basis
143 size_t const basisSize = basis.size();
144 std::vector<std::unique_ptr<Poly> > polys;
145 for (size_t gen = 0; gen != basisSize; ++gen)
146 polys.push_back(make_unique<Poly>(*basis.getPoly(gen)));
147 insertPolys(polys);
148}

157void ClassicGBAlg::insertPolys(
158 std::vector<std::unique_ptr<Poly> >& polynomials
159) {
160 if (!mUseAutoTopReduction) {
161 for (auto it = polynomials.begin(); it != polynomials.end(); ++it) {
162 MATHICGB_ASSERT(it->get() != 0);
163 if ((*it)->isZero())
164 continue;
165 if (mBasis.divisor((*it)->leadMono()) != static_cast<size_t>(-1)) {
166 *it = mReducer.classicReduce(**it, mBasis);
167 if ((*it)->isZero())
168 continue;
169 }
170
171 mBasis.insert(std::move(*it));
172 mSPairs.addPairs(mBasis.size() - 1);
173 }
174 polynomials.clear();
175 return;
176 }
177
178 std::vector<size_t> toRetire;
179 std::vector<std::unique_ptr<Poly>> toReduce;
180 std::vector<std::unique_ptr<Poly>> toInsert;
181 std::swap(toInsert, polynomials);
182
183 while (!toInsert.empty()) {
184 // todo: sort by lead term to avoid insert followed by immediate
185 // removal.
186
187 // insert polynomials from toInsert with minimal lead term and
188 // extract those from the basis that become non-minimal.
189 for (auto it = toInsert.begin(); it != toInsert.end(); ++it) {
190 MATHICGB_ASSERT(it->get() != 0);
191 if ((*it)->isZero())
192 continue;
193
194 // We check for a divisor from mBasis because a new reducer
195 // might have been added since we did the reduction or perhaps a
196 // non-reduced polynomial was passed in.
197 if (mBasis.divisor((*it)->leadMono()) != static_cast<size_t>(-1))
198 toReduce.push_back(std::move(*it));
199 else {
200 MATHICGB_IF_STREAM_LOG(GBInsert) {
201 stream << "Inserting basis element " << mBasis.size() << ": ";
202 MathicIO<>().writePoly(**it, true, stream);
203 stream << '\n';
204 };
205 mBasis.insert(std::move(*it));
206 MATHICGB_ASSERT(toRetire.empty());
207 mSPairs.addPairsAssumeAutoReduce(mBasis.size() - 1, toRetire);
208 for (auto r = toRetire.begin(); r != toRetire.end(); ++r)
209 toReduce.push_back(mBasis.retire(*r));
210 toRetire.clear();
211 }
212 }
213 toInsert.clear();
214 MATHICGB_ASSERT(toRetire.empty());
215
216 // reduce everything in toReduce
217 if (!toReduce.empty()) {
218 mReducer.classicReducePolySet(toReduce, mBasis, toInsert);
219 toReduce.clear();
220 }
221 }
222
223 MATHICGB_ASSERT(toRetire.empty());
224 MATHICGB_ASSERT(toInsert.empty());
225 MATHICGB_ASSERT(toReduce.empty());
226}

228void ClassicGBAlg::insertReducedPoly(
229 std::unique_ptr<Poly> polyToInsert
230) {
231 MATHICGB_ASSERT(polyToInsert.get() != 0);
232 if (polyToInsert->isZero())
233 return;
234 MATHICGB_ASSERT(mBasis.divisor(polyToInsert->leadMono()) ==
235 static_cast<size_t>(-1));
236
237 if (tracingLevel > 20) {
238 std::cerr << "inserting basis element " << mBasis.size() << ": ";
239 if (tracingLevel > 100) {
240 MathicIO<>().writePoly(*polyToInsert, false, std::cerr);
241 std::cerr << std::endl;
242 } else {
243 mRing.printMonomialFrobbyM2Format
244 (std::cerr, Monoid::toOld(polyToInsert->leadMono()));
245 if (polyToInsert->termCount() > 1)
246 std::cerr << " + [...]";
247 std::cerr << std::endl;
248 }
249 }
250
251 if (!mUseAutoTopReduction) {
252 size_t const newGen = mBasis.size();
253 mBasis.insert(std::move(polyToInsert));
254 mSPairs.addPairs(newGen);
255 return;
256 }
257
258 std::vector<size_t> toRetireAndReduce;
259 std::vector<Poly*> toReduce;
260
261 try {
262 do {
263 // reduce polynomial and insert into basis
264 {
265 std::unique_ptr<Poly> reduced;
266 if (toReduce.empty()) // if first iteration
267 reduced = std::move(polyToInsert);
268 else {
269 reduced = mReducer.classicReduce(*toReduce.back(), mBasis);
270 if (tracingLevel > 20) {
271 if (reduced->isZero()) {
272 std::cerr << "auto-top-reduce cascade: "
273 "basis element reduced to zero."
274 << std::endl;
275 } else {
276 std::cerr << "auto-top-reduce cascade: "
277 "inserting reduced poly with lead term "
278 << std::endl;
279 mRing.printMonomialFrobbyM2Format
280 (std::cerr, Monoid::toOld(reduced->leadMono()));
281 std::cerr << '\n';
282 }
283 }
284 delete toReduce.back();
285 toReduce.pop_back();
286 }
287 if (reduced->isZero())
288 continue;
289 reduced->makeMonic();
290 mBasis.insert(std::move(reduced));
291 }
292
293 // form S-pairs and retire basis elements that become top reducible.
294 const size_t newGen = mBasis.size() - 1;
295 MATHICGB_ASSERT(toRetireAndReduce.empty());
296 mSPairs.addPairsAssumeAutoReduce(newGen, toRetireAndReduce);
297 for (std::vector<size_t>::const_iterator it = toRetireAndReduce.begin();
298 it != toRetireAndReduce.end(); ++it) {
299 toReduce.push_back(0); // allocate space in vector before .release()
300 toReduce.back() = mBasis.retire(*it).release();
301 }
302 toRetireAndReduce.clear();
303 } while (!toReduce.empty());
304 } catch (...) {
305 for (std::vector<Poly*>::iterator it = toReduce.begin();
306 it != toReduce.end(); ++it)
307 delete *it;
308 throw;
309 }
310 MATHICGB_ASSERT(toReduce.empty());
311 MATHICGB_ASSERT(toRetireAndReduce.empty());
312}

314void ClassicGBAlg::computeGrobnerBasis() {
315 size_t counter = 0;
316 mTimer.reset();
317
318 if (mUseAutoTailReduction)
319 autoTailReduce();
320
321 while (!mSPairs.empty()) {
322 if (mCallback != nullptr && !mCallback())
323 break;
324
325 step();
326 if (mBreakAfter != 0 && mBasis.size() > mBreakAfter) {
327 std::cerr
328 << "Stopping Grobner basis computation due to reaching limit of "
329 << mBreakAfter << " basis elements." << std::endl;
330 break;
331 }
332 if (mPrintInterval != 0 && (++counter % mPrintInterval) == 0)
333 printStats(std::cerr);
334 }
335 if (mPrintInterval != 0)
336 printStats(std::cerr);

340 if (!mBasis.retired(i))
341 mBasis.replaceSameLeadTerm

346void ClassicGBAlg::step() {
347 MATHICGB_ASSERT(!mSPairs.empty());
348 if (tracingLevel > 30)
349 std::cerr << "Determining next S-pair" << std::endl;
350
351 MATHICGB_ASSERT(mSPairGroupSize >= 1);
352 std::vector<std::pair<size_t, size_t> > spairGroup;
353 exponent w = 0;
354 for (unsigned int i = 0; i < mSPairGroupSize; ++i) {
355 auto p = mSPairs.pop(w);
356 if (p.first == static_cast<size_t>(-1)) {
357 MATHICGB_ASSERT(p.second == static_cast<size_t>(-1));
358 break; // no more S-pairs
359 }
360 MATHICGB_ASSERT(p.first != static_cast<size_t>(-1));
361 MATHICGB_ASSERT(p.second != static_cast<size_t>(-1));
362 MATHICGB_ASSERT(!mBasis.retired(p.first));
363 MATHICGB_ASSERT(!mBasis.retired(p.second));
364
365 spairGroup.push_back(p);
366 }
367 if (spairGroup.empty())
368 return; // no more s-pairs
369 std::vector<std::unique_ptr<Poly>> reduced;
370
371 // w is the negative of the degree of the lcm's of the chosen spairs
372 MATHICGB_LOG(SPairDegree) <<
373 spairGroup.size() << " pairs in degree " << -w << std::endl;
374
375 mReducer.classicReduceSPolySet(spairGroup, mBasis, reduced);
376
377 // sort the elements to get deterministic behavior. The order will change
378 // arbitrarily when running multithreaded. Also, if preferring older
379 // reducers, it is of benefit to break ties by preferring the sparser
380 // reducer. Age does not have ties, since each element has a distinct
381 // index, but if they all come from the same matrix then there is
382 // really nothing to distinguish them - the relative age is arbitrarily
383 // chosen. If we order sparsest-first, we will effectively make the
384 // arbitrary choice among reducers from the same matrix in favor of sparser
385 // reducers.
386 auto order = [&](
387 const std::unique_ptr<Poly>& a,
388 const std::unique_ptr<Poly>& b
389 ) {
390 const auto aTermCount = a->termCount();
391 const auto bTermCount = b->termCount();
392 if (aTermCount < bTermCount)
393 return true;
394 if (aTermCount > bTermCount)
395 return false;
396 auto bIt = b->begin();
397 const auto aEnd = a->end();
398 for (auto aIt = a->begin(); aIt != aEnd; ++aIt, ++bIt) {
399 const auto monoCmp =
400 mRing.monoid().compare(aIt.mono(), bIt.mono());
401 if (monoCmp == LT)
402 return true;
403 if (monoCmp == GT)
404 return false;
405 if (aIt.coef() < bIt.coef())
406 return true;
407 if (aIt.coef() > bIt.coef())
408 return false;
409 }
410 return false;
411 };
412 std::sort(reduced.begin(), reduced.end(), order);
413
414 insertPolys(reduced);
415 if (mUseAutoTailReduction)
416 autoTailReduce();
417}

419void ClassicGBAlg::autoTailReduce() {
420 MATHICGB_ASSERT(mUseAutoTailReduction);
421
422 for (size_t i = 0; i < mBasis.size(); ++i) {
423 if (mBasis.retired(i))
424 continue;
425 if (mBasis.usedAsReducerCount(i) < 1000)
426 continue;
427 mBasis.replaceSameLeadTerm
428 (i, mReducer.classicTailReduce(mBasis.poly(i), mBasis));
429 }
430}

661Basis computeGBClassicAlg(
662 Basis&& inputBasis,
663 ClassicGBAlgParams params
664) {
665 ClassicGBAlg alg(
666 inputBasis,
667 *params.reducer,
668 params.monoLookupType,
669 params.preferSparseReducers,
670 params.sPairQueueType
671 );
672 alg.setBreakAfter(params.breakAfter);
