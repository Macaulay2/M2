77template <typename Sorter>
78long QuickSorter<Sorter>::sort_partition(long lo, long hi)
79{
80 value pivot = elems[lo];
81 long i = lo - 1;
82 long j = hi + 1;
83 for (;;)
84 {
85 do
86 {
87 j--;
88 }
89 while (M->compare(elems[j], pivot) < 0);
90 do
91 {
92 i++;
93 }
94 while (M->compare(elems[i], pivot) > 0);
95
96 if (i < j)
97 {
98 value tmp = elems[j];
99 elems[j] = elems[i];
100 elems[i] = tmp;
101 }
102 else
103 return j;
104 }
105}

108void QuickSorter<Sorter>::sort(long lo, long hi)
109{

111 {
112 long q = sort_partition(lo, hi);
113 sort(lo, q);
114 sort(q + 1, hi);
115 }

119// Quicksort, also recursive, better pivot choices
120// But: is very poor if all elements are the same

123/***** macros create functional code *****/
124#define pivot_index() (begin + (end - begin) / 2)
125#define swap(a, b, t) ((t) = (a), (a) = (b), (b) = (t))

127template <typename Sorter>
128void QuickSorter<Sorter>::sort2(long begin, long end)
129{
130 value pivot;
131 value t; /* temporary variable for swap */
132 if (end > begin)
133 {
134 long l = begin + 1;
135 long r = end;
136 swap(elems[begin],
137 elems[pivot_index()],
138 t); /*** choose arbitrary pivot ***/
139 pivot = elems[begin];
140 while (l < r)
141 {
142 if (M->compare(elems[l], pivot) <= 0)
143 {
144 l++;
145 }
146 else
147 {
148 while (l < --r &&
149 M->compare(elems[r], pivot) >=
150 0) /*** skip superfluous swaps ***/
151 ;
152 swap(elems[l], elems[r], t);
153 }
154 }
155 l--;
156 swap(elems[begin], elems[l], t);
157 sort2(begin, l);
158 sort2(r, end);
159 }
160}

209#define SWAP(a, b) \
210 temp = (a); \
211 (a) = (b); \
212 (b) = temp;
213#define THRESH 6
214#define NSTACK 50

216template <typename Sorter>
217void QuickSorter<Sorter>::sortC()
218// (unsigned long n, float arr[])
219// array is: elems
220// length is: len
221{
222 unsigned long i, ir = len, j, k, l = 1, *istack;
223 int jstack = 0;
224 value a, temp;
225 long ncmps;
226 int maxstack = 0;
227 ncmps = 0;
228 istack = newarray_atomic(unsigned long, NSTACK);
229 for (;;)
230 {
231 if (ir - l < THRESH)
232 {
233 for (j = l + 1; j <= ir; j++)
234 {
235 a = elems[j];
236 for (i = j - 1; i >= l; i--)
237 {
238 ncmps++;
239 if (M->compare(elems[i], a) <= 0) break;
240 elems[i + 1] = elems[i];
241 }
242 elems[i + 1] = a;
243 }
244 if (jstack == 0) break;
245 ir = istack[jstack--];
246 l = istack[jstack--];
247 }
248 else
249 {
250 k = (l + ir) >> 1;
251 SWAP(elems[k], elems[l + 1])
252 ncmps++;
253 if (M->compare(elems[l], elems[ir]) > 0)
254 {
255 SWAP(elems[l], elems[ir])
256 }
257 ncmps++;
258 if (M->compare(elems[l + 1], elems[ir]) > 0)
259 {
260 SWAP(elems[l + 1], elems[ir])
261 }
262 ncmps++;
263 if (M->compare(elems[l], elems[l + 1]) > 0)
264 {
265 SWAP(elems[l], elems[l + 1])
266 }
267 i = l + 1;
268 j = ir;
269 a = elems[l + 1];
270 for (;;)
271 {
272 do
273 i++;
274 while (ncmps++, M->compare(elems[i], a) < 0);
275 do
276 j--;
277 while (ncmps++, M->compare(elems[j], a) > 0);
278 if (j < i) break;
279 SWAP(elems[i], elems[j]);
280 }
281 elems[l + 1] = elems[j];
282 elems[j] = a;
283 jstack += 2;
284 if (jstack > maxstack) maxstack = jstack;
285 if (jstack > NSTACK)
286 {
287 fprintf(stderr, "NSTACK too small in sort.\n");
288 exit(0);
289 }
290 if (ir - i + 1 >= j - l)
291 {
292 istack[jstack] = ir;
293 istack[jstack - 1] = i;
294 ir = j - 1;
295 }
296 else
297 {
298 istack[jstack] = j - 1;
299 istack[jstack - 1] = l;
300 l = i;
301 }
302 }
303 }
304 freemem(istack);
305 fprintf(stderr,
306 "quicksort: len = %ld ncmps = %ld 2*depth = %d\n",
307 len,
308 ncmps,
309 maxstack);
310}

319template <typename Sorter>
320void QuickSorter<Sorter>::sortD()
321{
322 unsigned long i, ir, j, l;
323 value rra;
324 long ncmps;
325
326 ncmps = 0;
327 if (len < 2) return;
328 l = (len >> 1) + 1;
329 ir = len;
330 for (;;)
331 {
332 if (l > 1)
333 {
334 rra = elems[--l];
335 }
336 else
337 {
338 rra = elems[ir];
339 elems[ir] = elems[1];
340 if (--ir == 1)
341 {
342 elems[1] = rra;
343 break;
344 }
345 }
346 i = l;
347 j = l + l;
348 while (j <= ir)
349 {
350 if (j < ir)
351 {
352 ncmps++;
353 if (M->compare(elems[j], elems[j + 1]) < 0) j++;
354 }
355 ncmps++;
356 if (M->compare(rra, elems[j]) < 0)
357 {
358 elems[i] = elems[j];
359 i = j;
360 j <<= 1;
361 }
362 else
363 j = ir + 1;
364 }
365 elems[i] = rra;
366 }
367 fprintf(stderr, "hpsort: %ld, %ld\n", len, ncmps);
368}

375{
376 QuickSorter S(M0, elems0, len0);
377 int typ = 1;
378 clock_t begin_time = clock();
379
380 switch (typ)

392
