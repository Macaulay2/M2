56struct Monom
57// Format for monomials:
58 // A monomial is an array of ints, the first of which is the length of that array (including length field).
59 // e.g. [2 0] (is currently the empty monomial, but this class knows nothing about specific monomials.
60 // each monomial is of the form:
61 // <length: n+2> <degree: (currently) n> <var1> <var2> ... <varn>
62 // e.g. xyxy: 6 4 0 1 0 1
63 // xy23x: 27 25 0 1 1 ... 1 0
64 // 2 monomials: xzx, xy (dot is only there for readability)
65 // 5 3 0 2 0 . 4 2 0 1
66 // TODO: There are now weights in the monomial (which only the monoid knows about),
67 // so this needs to be updated.
68{
69 Monom() : mValue(nullptr) {}

71 // const int* operator*() const { return mValue; }

78 const int* begin() const { return mValue; }
79 const int* end() const { return mValue + *mValue; }

143 static int compare(const ModuleMonom& m1, const ModuleMonom& m2)
144 {
145 if (m1[2] > m2[2]) return GT;
146 if (m1[2] < m2[2]) return LT;
147 if (m1[3] > m2[3]) return GT;
148 if (m1[3] < m2[3]) return LT;
149 // at this stage, they have the same degree, so use lex order
150 for (int j = 4; j < m1[0]; j++)
151 {
152 if (m1[j] > m2[j]) return LT;
153 if (m1[j] < m2[j]) return GT;
154 }
155 // if we are here, the monomials are the same.
156 return EQ;
157 }

159 bool operator==(const ModuleMonom &rhs) const
160 {
161 if (mValue[0] != rhs[0]) return false;
162 for (int i=2; i < mValue[0]; ++i)
163 if (mValue[i] != rhs[i]) return false;
164 return true;
165 }
166private:
167 void setHashValue() const
168 {
169 int result = 0;
170 int* end = mValue + *mValue;
171 for (auto i = mValue+3; i < end; ++i)
172 result = 17*result + *i;
173 const_cast<int*>(mValue)[2] = result;
174 }

181inline ModuleMonom monomToModuleMonom(const Monom& a, int comp, std::pair<int*, int*> allocated_result)
182{
183 assert(allocated_result.second-allocated_result.first >= ModuleMonom::sizeOfCorrespondingModuleMonom(a));
184 auto begin = allocated_result.first;
185 begin[0] = ModuleMonom::sizeOfCorrespondingModuleMonom(a);
186 begin[1] = 0; // index
187 begin[2] = 0; // hashval
188 begin[3] = comp;
189 std::copy(a.begin()+1, a.end(), begin+4);
190 return ModuleMonom(begin);
191}
