43bool almostEqual(const M2::ARingCC& C,
44 unsigned long nbits,
45 const M2::ARingCC::ElementType& a,
46 const M2::ARingCC::ElementType& b)
47{
48 M2::ARingRR::ElementType epsilon = pow(2, static_cast<double>(-nbits));
49 M2::ARingRR::ElementType d;
50 M2::ARingCC::ElementType c;
51 C.subtract(c, a, b);
52 C.abs(d, c);
53 return C.real_ring().compare_elems(d, epsilon) < 0;
54}

186TEST(ARingCC, axioms)
187{
188 M2::ARingCC C;
189 auto nbits = C.get_precision();
190 ARingElementGenerator<M2::ARingCC> gen(C);
191 M2::ARingCC::ElementType a, b, c, d, e;
192 C.init(a);
193 C.init(b);
194 C.init(c);
195 C.init(d);
196 C.init(e);
197 for (int i = 0; i < ntrials; i++)
198 {
199 gen.nextElement(a);
200 gen.nextElement(b);
201 gen.nextElement(c);
202 // Test commutativity
203 // test: a*b = b*a
204 // test: a+b == b+a
205 C.add(d, a, b);
206 C.add(e, b, a);
207 EXPECT_TRUE(almostEqual(C, nbits - 2, d, e));
208 C.mult(d, a, b);
209 C.mult(e, b, a);
210 EXPECT_TRUE(almostEqual(C, nbits - 2, d, e));
211
212 // Test associativity
213 // test: a+(b+c) == (a+b)+c
214 // test: a*(b*c) == (a*b)*c
215 C.add(e, b, c);
216 C.add(d, a, e); // a+(b+c)
217 C.add(e, a, b);
218 C.add(e, e, c); // (a+b)+c
219 EXPECT_TRUE(almostEqual(C, nbits - 6, d, e));
220 C.mult(e, b, c);
221 C.mult(d, a, e); // a*(b*c)
222 C.mult(e, a, b);
223 C.mult(e, e, c); // (a*b)*c
224 EXPECT_TRUE(almostEqual(C, nbits - 6, d, e));
225
226 // Test distributivity
227 // test: a*(b+c) == a*b + a*c
228 C.add(e, b, c);
229 C.mult(d, a, e); // a*(b+c)
230 C.mult(b, a, b);
231 C.mult(c, a, c);
232 C.add(e, b, c); // a*b + a*c
233 EXPECT_TRUE(almostEqual(C, nbits - 6, d, e));
234 }
235 C.clear(e);
236 C.clear(d);
237 C.clear(c);
238 C.clear(b);
239 C.clear(a);
240}

277 }
