98int GBKernelComputation::calc()
99{
100 // First find the skeleton
101 for (int i = 0; i < gb.size(); i++) new_pairs(i);
102
103 // Debug code
104 GBMatrix *mm = new GBMatrix(G);
105 for (int p = 0; p < syzygies.size(); p++)
106 mm->append(GR->gbvector_copy(syzygies[p]));
107 buffer o;
108 Matrix *m = mm->to_matrix();
109 if (M2_gbTrace >= 5)
110 {
111 o << "skeleton = " << newline;
112 m->text_out(o);
113 emit(o.str());
114 }
115
116 // Sort the skeleton now?
117
118 // Now reduce each one of these elements
119 for (int j = 0; j < syzygies.size(); j++)
120 {
121 gbvector *v = s_pair(syzygies[j]);
122 reduce(v, syzygies[j]);
123 }
124 return COMP_DONE;
125}

156void GBKernelComputation::strip_gb(const GBMatrix *m)
157{
158 const gc_vector<gbvector*> &g = m->elems;
159 int i;
160 int *components = newarray_atomic_clear(int, F->rank());
161 for (i = 0; i < g.size(); i++)
162 if (g[i] != nullptr) components[g[i]->comp - 1]++;
163
164 for (i = 0; i < g.size(); i++)
165 {
166 gbvector head;
167 gbvector *last = &head;
168 for (gbvector *v = g[i]; v != nullptr; v = v->next)
169 if (components[v->comp - 1] > 0)
170 {
171 gbvector *t = GR->gbvector_copy_term(v);
172 last->next = t;
173 last = t;
174 }
175 last->next = nullptr;
176 gb.push_back(head.next);
177 }
178 for (i = 0; i < F->rank(); i++) mi.push_back(new MonomialIdeal(R));
179 freemem(components);
180}

182void GBKernelComputation::new_pairs(int i)
183// Create and insert all of the pairs which will have lead term 'gb[i]'.
184// This also places 'in(gb[i])' into the appropriate monomial ideal
185{
186 gc_vector<Bag*> elems;
187 gc_vector<int> vp; // This is 'p'.
188 gc_vector<int> thisvp;
189
190 if (SF)
191 {
192 monomial PAIRS_mon = ALLOCATE_MONOMIAL(monom_size);
193 SF->schreyer_down(gb[i]->monom, gb[i]->comp - 1, PAIRS_mon);
194 M->to_varpower(PAIRS_mon, vp);
195 }
196 else
197 M->to_varpower(gb[i]->monom, vp);
198
199 // First add in syzygies arising from exterior variables
200 // At the moment, there are none of this sort.
201
202 if (R->is_skew_commutative())
203 {
204 exponents_t find_pairs_exp = newarray_atomic(int, M->n_vars());
205
206 varpower::to_expvector(M->n_vars(), vp.data(), find_pairs_exp);
207 for (int w = 0; w < R->n_skew_commutative_vars(); w++)
208 if (find_pairs_exp[R->skew_variable(w)] > 0)
209 {
210 thisvp.resize(0);
211 varpower::var(w, 1, thisvp);
212 Bag *b = new Bag(static_cast<void *>(nullptr), thisvp);
213 elems.push_back(b);
214 }
215
216 freemem(find_pairs_exp);
217 }
218
219 // Second, add in syzygies arising from the base ring, if any
220 // The baggage of each of these is NULL
221 if (R->is_quotient_ring())
222 {
223 const MonomialIdeal *Rideal = R->get_quotient_monomials();
224 for (Bag& a : *Rideal)
225 {
226 // Compute (P->quotient_ideal->monom : p->monom)
227 // and place this into a varpower and Bag, placing
228 // that into 'elems'
229 thisvp.resize(0);
230 varpower::quotient(a.monom().data(), vp.data(), thisvp);
231 if (varpower::is_equal(a.monom().data(), thisvp.data()))
232 continue;
233 Bag *b = new Bag(static_cast<void *>(nullptr), thisvp);
234 elems.push_back(b);
235 }
236 }
237
238 // Third, add in syzygies arising from previous elements of this same level
239 // The baggage of each of these is their corresponding res2_pair
240
241 MonomialIdeal *mi_orig = mi[gb[i]->comp - 1];
242 for (Bag& a : *mi_orig)
243 {
244 Bag *b = new Bag();
245 varpower::quotient(a.monom().data(), vp.data(), b->monom());
246 elems.push_back(b);
247 }
248
249 // Make this monomial ideal, and then run through each minimal generator
250 // and insert into the proper degree. (Notice that sorting does not
251 // need to be done yet: only once that degree is about to begin.
252
253 mi_orig->insert_minimal(new Bag(i, vp));
254
255 MonomialIdeal *new_mi = new MonomialIdeal(R, elems);
256
257 monomial m = M->make_one();
258 for (Bag& a : *new_mi)
259 {
260 M->from_varpower(a.monom().data(), m);
261 M->mult(m, gb[i]->monom, m);
262
263 gbvector *q = make_syz_term(
264 GR->get_flattened_coefficients()->from_long(1), m, i + 1);
265 syzygies.push_back(q);
266 }
267}

273bool GBKernelComputation::find_ring_divisor(const_exponents exp,
274 const gbvector *&result)
275// If 'exp' is divisible by a ring lead term, then 1 is returned,
276// and result is set to be that ring element.
277// Otherwise 0 is returned.
278{
279 if (!R->is_quotient_ring()) return false;
280 Bag *b;
281 if (!R->get_quotient_monomials()->search_expvector(exp, b)) return false;
282 int index = b->basis_elem();
283 result = R->quotient_gbvector(index);
284 return true;
285}

287int GBKernelComputation::find_divisor(const MonomialIdeal *this_mi,
288 const_exponents exp,
289 int &result)
290{
291 // Find all the possible matches, use some criterion for finding the best...
292 gc_vector<Bag*> bb;
293 this_mi->find_all_divisors(exp, bb);
294 int ndivisors = bb.size();
295 if (ndivisors == 0) return 0;
296 result = bb[0]->basis_elem();
297 // Now search through, and find the best one. If only one, just return it.
298 if (M2_gbTrace >= 5)
299 if (this_mi->size() > 1)
300 {
301 buffer o;
302 o << ":" << this_mi->size() << "." << ndivisors << ":";
303 emit(o.str());
304 }
305 if (ndivisors == 1)
306 {
307 if (this_mi->size() == 1)
308 n_ones++;
309 else
310 n_unique++;
311 return 1;
312 }
313 n_others++;
314
315 int lowest = result;
316 for (int i = 1; i < ndivisors; i++)
317 {
318 int p = bb[i]->basis_elem();
319 if (p < lowest) lowest = p;
320 }
321 result = lowest;
322 return ndivisors;
323}

325gbvector *GBKernelComputation::s_pair(gbvector *gsyz)
326{
327 gbvector *result = nullptr;
328 monomial si = M->make_one();
329 for (gbvector *f = gsyz; f != nullptr; f = f->next)
330 {
331 SG->schreyer_down(f->monom, f->comp - 1, si);
332 gbvector *h = GR->mult_by_term(F, gb[f->comp - 1], f->coeff, si, 0);
333 wipe_unneeded_terms(h);
334 GR->gbvector_add_to(F, result, h);
335 }
336 M->remove(si);
337 return result;
338}

340void GBKernelComputation::wipe_unneeded_terms(gbvector *&f)
341{
342 // Remove every term of f (except the lead term)
343 // which is NOT divisible by an element of mi.
344 exponents_t exp = newarray_atomic(int, GR->n_vars());
345 // int nterms = 1;
346 // int nsaved = 0;
347 gbvector *g = f;
348 while (g->next != nullptr)
349 {
350 // First check to see if the term g->next is in the monideal
351 // nterms++;
352 Bag *b;
353 GR->gbvector_get_lead_exponents(F, g->next, exp);
354 if (mi[g->next->comp - 1]->search_expvector(exp, b))
355 {
356 // Want to keep the monomial
357 g = g->next;
358 }
359 else
360 {
361 // Want to dump this term
362 // nsaved++;
363 gbvector *tmp = g->next;
364 g->next = tmp->next;
365 tmp->next = nullptr;
366 GR->gbvector_remove(tmp);
367 }
368 }

371// {

373// o << "[" << nterms << ",s" << nsaved << "]";

379void GBKernelComputation::reduce(gbvector *&f, gbvector *&fsyz)
380{
381 exponents_t REDUCE_exp = ALLOCATE_EXPONENTS(exp_size);
382 monomial REDUCE_mon = ALLOCATE_MONOMIAL(monom_size);
383
384 const Ring *gbringK = GR->get_flattened_coefficients();
385 ring_elem one = gbringK->from_long(1);
386 gbvector *lastterm = fsyz; // fsyz has only ONE term.
387 const gbvector *r;
388 int q;
389
390 int nhits = 0;
391 int nremoved = 0;
392 int max_len = GR->gbvector_n_terms(f);
393
394 int count = 0;
395 if (M2_gbTrace >= 4) emit_wrapped(",");
396
397 while (f != nullptr)
398 {
399 if (SF)
400 {
401 SF->schreyer_down(f->monom, f->comp - 1, REDUCE_mon);
402 M->to_expvector(REDUCE_mon, REDUCE_exp);
403 }
404 else
405 M->to_expvector(f->monom, REDUCE_exp);
406 if (find_ring_divisor(REDUCE_exp, r))
407 {
408 ring_elem u, v;
409 // Subtract off f, leave fsyz alone
410 M->divide(f->monom, r->monom, REDUCE_mon);
411 gbringK->syzygy(f->coeff, r->coeff, u, v);
412 gbvector *h = GR->mult_by_term(F, r, v, REDUCE_mon, f->comp);
413 GR->gbvector_add_to(F, f, h);
414 if (!gbringK->is_equal(u, one))
415 {
416 GR->gbvector_mult_by_coeff_to(fsyz, u);
417 GR->gbvector_mult_by_coeff_to(f, u);
418 gbringK->remove(u);
419 gbringK->remove(v);
420 }
421 total_reduce_count++;
422 count++;
423 }
424 else if (find_divisor(mi[f->comp - 1], REDUCE_exp, q))
425 {
426 ring_elem u, v;
427 gbringK->syzygy(f->coeff, gb[q]->coeff, u, v);
428 M->divide(f->monom, gb[q]->monom, REDUCE_mon);
429 gbvector *h = GR->mult_by_term(F, gb[q], v, REDUCE_mon, 0);
430 if (!gbringK->is_equal(u, one))
431 {
432 GR->gbvector_mult_by_coeff_to(fsyz, u);
433 GR->gbvector_mult_by_coeff_to(f, u);
434 }
435 int n1 = GR->gbvector_n_terms(h);
436 wipe_unneeded_terms(h);
437 int n2 = GR->gbvector_n_terms(h);
438 nremoved += (n1 - n2);
439 lastterm->next = make_syz_term(v, f->monom, q + 1); // grabs v.
440 lastterm = lastterm->next;
441 gbringK->remove(u);
442 gbringK->remove(v);
443 int n3 = GR->gbvector_n_terms(f);
444 GR->gbvector_add_to(F, f, h);
445 int n4 = GR->gbvector_n_terms(f);
446 if (n4 > max_len) max_len = n4;
447 nhits +=
448 (n2 + n3 - n4 -
449 2); // the -2 is to avoid counting the lead term cancellations.
450 total_reduce_count++;
451 count++;
452 }
453 else
454 {
455 // To get here is an ERROR!
456 f = f->next; // Just to not have an infinite loop
457 emit_line(
458 "error in Schreyer reduction: element does not reduce to zero!");
459 }
460 }
461
462 if (M2_gbTrace >= 4)
463 {
464 buffer o;
465 o << count;
466 o << "[" << max_len << " " << nhits << " " << nremoved << "]";
467 emit_wrapped(o.str());
468 }
469}

471void GBKernelComputation::geo_reduce(gbvector *&f, gbvector *&fsyz)
472{
473 exponents_t REDUCE_exp = ALLOCATE_EXPONENTS(exp_size);
474 monomial REDUCE_mon = ALLOCATE_MONOMIAL(monom_size);
475
476 const Ring *gbringK = GR->get_flattened_coefficients();
477 ring_elem one = gbringK->from_long(1);
478 gbvector *lastterm = fsyz; // fsyz has only ONE term.
479 const gbvector *r;
480 gbvectorHeap fb(GR, F);
481 fb.add(f);
482 f = nullptr;
483 const gbvector *lead;
484 int q;
485
486 int count = 0;
487 if (M2_gbTrace >= 4) emit_wrapped(",");
488
489 while ((lead = fb.get_lead_term()) != nullptr)
490 {
491 if (SF)
492 {
493 SF->schreyer_down(lead->monom, lead->comp - 1, REDUCE_mon);
494 M->to_expvector(REDUCE_mon, REDUCE_exp);
495 }
496 else
497 M->to_expvector(lead->monom, REDUCE_exp);
498 if (find_ring_divisor(REDUCE_exp, r))
499 {
500 ring_elem u, v;
501 // Subtract off f, leave fsyz alone
502 M->divide(lead->monom, r->monom, REDUCE_mon);
503
504 gbringK->syzygy(f->coeff, r->coeff, u, v);
505 gbvector *h = GR->mult_by_term(F, r, v, REDUCE_mon, f->comp);
506 fb.add(h);
507 if (!gbringK->is_equal(u, one))
508 {
509 GR->gbvector_mult_by_coeff_to(fsyz, u);
510 GR->gbvector_mult_by_coeff_to(f, u);

512 gbringK->remove(v);
513 }
514
515 fb.add(h);
516 total_reduce_count++;
517 count++;
518 }
519 else if (find_divisor(mi[lead->comp - 1], REDUCE_exp, q))
520 {
521 ring_elem u, v;

528 GR->gbvector_mult_by_coeff_to(f, u);

531 lastterm->next = make_syz_term(v, lead->monom, q + 1); // grabs v.
532 lastterm = lastterm->next;

534 total_reduce_count++;

536 }
