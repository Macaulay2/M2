230 inline void MonomLex::MultiplyBy(MonomLex::Integer var)
231 {
232 //inserted variable is the only one
233 if (!ListHead)
234 {
235 ListHead = new VarsListNode();
236 ListHead->Value = var;
237 ++TotalDegree;
238 }
239 else
240 {
241 VarsListNode* position = Find(var);
242 //inserted variable is the eldest one
243 if (!position)
244 {
245 position = new VarsListNode();
246 position->Value = var;
247 position->Next = ListHead;
248 ListHead = position;
249 ++TotalDegree;
250 }
251 //all other cases
252 else if(position->Value != var)
253 {
254 VarsListNode* newNode = new VarsListNode();
255 newNode->Value = var;
256 newNode->Next = position->Next;
257 position->Next = newNode;
