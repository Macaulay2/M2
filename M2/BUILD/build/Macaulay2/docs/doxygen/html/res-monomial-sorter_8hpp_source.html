120class ResMonomialSorter
121{
122private:
123 const Monoid& mMonoid;
124 const ResMonoid& mResMonoid;
125 const ResSchreyerOrder& mSchreyerOrder;
126 const std::vector<res_packed_monomial>& mColumns;
127
128 long mNumComparisons;
129 memt::Arena mArena;
130 std::vector<int*> mMonoms; // each monom: [tiebreaker, basecomp, followed by totalmon]
131 std::vector<int> mPositions;
132public:
133 ResMonomialSorter(const Monoid& M,
134 const ResMonoid& resMonoid,
135 const ResSchreyerOrder& S, // order at level-1 in free res
136 const std::vector<res_packed_monomial>& columns // at level.
137 ) :
138 mMonoid(M),
139 mResMonoid(resMonoid),
140 mSchreyerOrder(S),
141 mColumns(columns),
142 mNumComparisons(0)
143 {
144 }
145
146 void setMonoms()
147 {
148 for (int i=0; i<mColumns.size(); i++)
149 {
150 std::pair<int*, int*> mon = mArena.allocArrayNoCon<int>(mMonoid.monomial_size() + 2);
151 toMonomial(mColumns[i], mon);
152 mMonoms.push_back(mon.first);
153 }
154 }
155
156 bool ordered()
157 {
158 setMonoms();
159 MonomialSorterObject C(mMonoid, mMonoms);
160 for (int i=1; i<mMonoms.size(); i++)
161 {
162 if (not C(i-1,i)) return false;
163 }
164 return true;
165 }
166
167 std::vector<int> sort()
168 {
169 setMonoms();
170
171 std::vector<int> result;
172
173 for (int i=0; i<mColumns.size(); i++)
174 result.push_back(i);
175
176 MonomialSorterObject C(mMonoid, mMonoms);
177 C.resetNumComparisons();
178
179 std::stable_sort(result.begin(), result.end(), C);
180
181 mNumComparisons = C.numComparisons();
182 return result;
183 }
184
185 long numComparisons() const { return mNumComparisons; }
186private:
187 void toMonomial(res_packed_monomial mon, std::pair<int*,int*> resultAlreadyAllocateds)
188 {
189 int comp, comp2;
190 int nvars = mMonoid.n_vars();
191 std::pair<int*, int*> exp = mArena.allocArrayNoCon<int>(nvars);
192 std::pair<int*, int*> exp2 = mArena.allocArrayNoCon<int>(nvars);
193 mResMonoid.to_expvector(mon, exp.first, comp);
194 mResMonoid.to_expvector(mSchreyerOrder.mTotalMonom[comp], exp2.first, comp2);
195 exponents::mult(nvars, exp.first, exp2.first, exp2.first);
196 auto p = resultAlreadyAllocateds.first;
197 *p++ = mSchreyerOrder.mTieBreaker[comp];
198 *p++ = comp2;
199 mMonoid.from_expvector(exp2.first, p);
200 mArena.freeTop(exp2.first); // note: can only pop one at a time from an mt::Arena!
201 mArena.freeTop(exp.first);
202 }
203};
