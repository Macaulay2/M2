295class MyIdealConsumer : public Frobby::IdealConsumer, our_new_delete
296{
297 int nv; // The size of exponentVector coming from frobby
298 exponents_t exp;
299 MonomialIdeal *J;
300
301 public:
302 MyIdealConsumer(const PolynomialRing *R, int nv0) : nv(nv0)
303 {
304 J = new MonomialIdeal(R);
305 exp = newarray_atomic(int, nv);
306 }
307 ~MyIdealConsumer()
308 {
309 freemem(exp);
310 J = nullptr;
311 }
312 virtual void consume(mpz_ptr *exponentVector)
313 {
314 // insert into J. This is a minimal generator of J
315 for (int i = 0; i < nv; i++)
316 exp[i] = static_cast<int>(mpz_get_si(
317 exponentVector[i])); // overflow should not occur, as input fit
318
319 if (M2_gbTrace >= 5)
320 {
321 fprintf(stderr, "got ");
322 for (int j = 0; j < nv; j++) fprintf(stderr, "%d ", exp[j]);
323 fprintf(stderr, "\n");
324 }
325
326 Bag *b = new Bag();
327 varpower::from_expvector(nv, exp, b->monom());
328 J->insert_minimal(b);
329 }
330 MonomialIdeal *result() { return J; }
331};

333static MonomialIdeal *FrobbyAlexanderDual(const MonomialIdeal *I,
334 const mpz_t *topvec)
335{
336 int nv = I->topvar() + 1;
337 exponents_t exp = newarray_atomic(int, nv);
338 Frobby::Ideal F(nv);
339 for (Bag& b : *I)
340 {
341 varpower::to_expvector(nv, b.monom().data(), exp);
342
343 if (M2_gbTrace >= 4) fprintf(stderr, "adding ");
344 for (int j = 0; j < nv; j++)
345 {
346 if (M2_gbTrace >= 4) fprintf(stderr, "%d ", exp[j]);
347 F.addExponent(exp[j]);
348 }
349 if (M2_gbTrace >= 4) fprintf(stderr, "\n");
350 }
351
352 // Now create the consumer object, and call Frobby
353 MyIdealConsumer M(I->get_ring(), nv);
354 Frobby::alexanderDual(F, topvec, M);
355 freemem(exp);
356 // Extract the answer as a MonomialIdeal
357 return M.result();
358}

360static MonomialIdeal *wrapperFrobbyAlexanderDual(const MonomialIdeal *I,
361 const M2_arrayint top)
362// Assumption: top is an array of at least the number of variables of I
363// whose v-th entry is at least as large as the v-th exp of any mingen of I
364{
365 // Create a Frobby Ideal containing I.
366 int nv = I->topvar() + 1;
367 if (nv == 0)
368 {
369 INTERNAL_ERROR("attempting to use frobby with zero variables");
370 return nullptr;
371 }
372
373 mpz_t *topvec = nullptr;
374 if (top->len > 0)
375 {
376 topvec = newarray(mpz_t, top->len);
377 for (int i = 0; i < top->len; i++)
378 mpz_init_set_si(topvec[i], top->array[i]);
379 }
380
381 MonomialIdeal *result = FrobbyAlexanderDual(I, topvec);
382
383 // Clean up
384 if (topvec != nullptr)
385 {
386 for (int i = 0; i < top->len; i++) mpz_clear(topvec[i]);
387 freemem(topvec);
388 }
389
390 return result;
391}

393static MonomialIdeal /* or null */ *alexDual(const MonomialIdeal *I,
394 const M2_arrayint top,
395 int strategy)
396{
397 if (I->topvar() < 0)
398 strategy =
399 1; // i.e. don't use frobby if there are no generators and/or variables
400 switch (strategy)
401 {
402 case 0:
403 if (M2_gbTrace >= 1) emit_line(" -- [Alexander dual: frobby]");
404 return wrapperFrobbyAlexanderDual(I, top);
405 default:
406 if (M2_gbTrace >= 1) emit_line(" -- [Alexander dual: M2 monideal]");
407 return I->alexander_dual(top);
408 }
409}
