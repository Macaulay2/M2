51struct ReducedGB_Field_Local_sorter
52{
53 GBRing *R;
54 const FreeModule *F;
55 const VECTOR(POLY) & gb;
56 std::vector<int> degs;
57 ReducedGB_Field_Local_sorter(GBRing *R0,
58 const FreeModule *F0,
59 const VECTOR(POLY) & gb0)
60 : R(R0), F(F0), gb(gb0)
61 {
62 auto M = R->get_flattened_monoid();
63 for (size_t i = 0; i < gb0.size(); i++)
64 {
65 gbvector *f = gb0[i].f;
66 degs.push_back(M->simple_degree(f->monom));
67 }
68 }
69 bool operator()(int xx, int yy)
70 {
71 // this is the < operation
72 if (degs[xx] < degs[yy]) return true;
73 if (degs[xx] > degs[yy]) return false;
74 gbvector *x = gb[xx].f;
75 gbvector *y = gb[yy].f;
76 return R->gbvector_compare(F, x, y) == LT;
77 }
78};

80void ReducedGB_Field_Local::minimalize(const VECTOR(POLY) & polys0,
81 bool auto_reduced)
82{
83 // First sort these elements via increasing lex order (or monomial order?)
84 // Next insert minimal elements into T, and polys
85
86 VECTOR(int) positions;
87 positions.reserve(polys0.size());
88
89 for (int i = 0; i < polys0.size(); i++) positions.push_back(i);
90
91 // displayElements("-- before sort --", R, polys0, [](auto& g) { return g.f;
92 // } );
93
94 std::stable_sort(positions.begin(),
95 positions.end(),
96 ReducedGB_Field_Local_sorter(R, F, polys0));
97
98 // VECTOR(gbvector*) sorted_elements_debug_only;
99 // for (int i=0; i<positions.size(); i++)
100 // sorted_elements_debug_only.push_back(polys0[positions[i]].f);
101 // displayElements("-- after sort --", R, sorted_elements_debug_only,
102 // [](auto& g) { return g; } );
103
104 // Now loop through each element, and see if the lead monomial is in T.
105 // If not, add it in , and place element into 'polys'.
106
107 for (VECTOR(int)::iterator i = positions.begin(); i != positions.end(); i++)
108 {
109 Bag *not_used;
110 gbvector *f = polys0[*i].f;
111 exponents_t e = R->exponents_make();
112 R->gbvector_get_lead_exponents(F, f, e);
113 if ((!Rideal || !Rideal->search_expvector(e, not_used)) &&
114 T->find_divisors(1, e, f->comp) == 0)
115 {
116 // Keep this element
117
118 POLY h;
119 ring_elem junk;
120
121 h.f = R->gbvector_copy(f);
122 h.fsyz = R->gbvector_copy(polys0[*i].fsyz);
123
124 if (false and auto_reduced)
125 remainder(h, false, junk); // This auto-reduces h.
126
127 R->gbvector_remove_content(h.f, h.fsyz);
128
129 T->insert(e, f->comp, INTSIZE(polys));
130 polys.push_back(h);
131 }
132 else
133 R->exponents_delete(e);
134 }
135
136 for (int i = 0; i < polys.size(); i++)
137 {
138 int f_lead_wt;
139 gbvector *f = polys[i].f;
140 int d = wt->gbvector_weight(f, f_lead_wt);
141 int a = d - f_lead_wt;
142
143 divisor_info t;
144 t.g = polys[i];
145 t.size = R->gbvector_n_terms(f);
146 t.alpha = a;
147 gb_elems.push_back(t);
148 }
149}

490void ReducedGB_Field_Local::remainder(POLY &f, bool use_denom, ring_elem &denom)
491{
492 if (M2_gbTrace >= 4) {
493 buffer o;
494 text_out(o);
495 emit(o.str());
496 }
497 if (f.f == nullptr) return;
498 T1 = MonomialTable::make(R->n_vars());
499 gbvector head;
500 gbvector *frem = &head;
501 frem->next = nullptr;
502 POLY h = f;
503 exponents_t h_exp = R->exponents_make();
504 int h_alpha, g_alpha;
505 int h_deg = wt->gbvector_weight(f.f);
506 while (!R->gbvector_is_zero(h.f))
507 {
508 if (M2_gbTrace == 3) emit_wrapped(".");
509 POLY g;
510 R->gbvector_get_lead_exponents(F, h.f, h_exp);
511 int h_comp = h.f->comp;
512
513 if (M2_gbTrace >= 4)
514 {
515 buffer o;
516 o << "\nreducing ";
517 R->gbvector_text_out(o, F, h.f);
518 emit(o.str());
519 }
520
521 if (find_good_divisor(h_exp,
522 h_comp,
523 h_deg,
524 h_alpha,
525 g,
526 g_alpha)) // sets these three values
527 {
528 if (M2_gbTrace >= 4)
529 {
530 buffer o;
531 o << " h_alpha " << h_alpha << " g_alpha "
532 << g_alpha; // << " reducing using ";
533 // R->gbvector_text_out(o,F,g.f);
534 // o << newline;
535 emit(o.str());
536 }
537 if (g_alpha > h_alpha)
538 {
539 if (head.next != nullptr)
540 {
541 // In this case, we can't reduce the tail without
542 // risking an infinite loop. So we declare ourselves done
543 // Attach the rest of h.f to frem
544 frem->next = h.f;
545 break;
546 }
547 // place h into T1, and store its (value,deg,alpha) values.
548 // store_in_table copies h
549 store_in_table(h, h_exp, h_comp, h_alpha);
550 if (M2_gbTrace == 3) emit_wrapped("x");
551 if (M2_gbTrace == 4) emit("\nstored result\n");
552 h_deg += g_alpha - h_alpha;
553 h_exp = R->exponents_make();
554 }
555 R->gbvector_reduce_lead_term(
556 F, Fsyz, head.next, h.f, h.fsyz, g.f, g.fsyz, use_denom, denom);
557 }
558 else
559 {
560 frem->next = h.f;
561 frem = frem->next;
562 h.f = h.f->next;
563 frem->next = nullptr;
564 }
565 }
566
567 f.f = head.next;
568 f.fsyz = h.fsyz;
569 R->exponents_delete(h_exp);
570 reset_table();
571}
