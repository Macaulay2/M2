164KBasis::KBasis(const Matrix *bottom,
165 const int *lo_degree0,
166 const int *hi_degree0,
167 std::vector<int> heftvec,
168 std::vector<int> varlist,
169 bool do_truncation0,
170 int limit0)
171 : bottom_matrix(bottom),
172 mHeftVector(heftvec),
173 mVariables(varlist),
174 do_truncation(do_truncation0),
175 weight_has_zeros(false),
176 limit(limit0),
177 lo_degree(lo_degree0),
178 hi_degree(hi_degree0),
179 kb_error(false)
180{
181 P = bottom->get_ring()->cast_to_PolynomialRing();
182 M = P->getMonoid();
183 D = P->get_degree_ring()->getMonoid();
184
185 if (lo_degree == nullptr && hi_degree == nullptr) { computation_type = KB_FULL; }
186 else if (mHeftVector.size() == 1) { computation_type = KB_SINGLE; }
187 else { computation_type = KB_MULTI; }
188
189 // Compute the (non-negative) weights of each of the variables in
190 // 'mVariables'.
191
192 var_wts = newarray_atomic(int, mVariables.size());
193 var_degs = newarray_atomic(int, mVariables.size() * mHeftVector.size());
194 int *exp =
195 newarray_atomic(int, D->n_vars()); // used to hold exponent vectors
196 int next = 0;
197 for (int i = 0; i < mVariables.size(); i++, next += mHeftVector.size())
198 {
199 int v = mVariables[i];
200 D->to_expvector(M->degree_of_var(v), exp);
201 var_wts[i] = exponents::weight(mHeftVector.size(), exp, mHeftVector);
202 if (var_wts[i] == 0) weight_has_zeros = true;
203 exponents::copy(mHeftVector.size(), exp, var_degs + next);
204 }
205 freemem(exp);
206
207 // Set the recursion variables
208 kb_exp = newarray_atomic_clear(int, P->n_vars());
209 kb_exp_weight = 0;
210
211 if (lo_degree != nullptr)
212 kb_target_lo_weight =
213 exponents::weight(mHeftVector.size(), lo_degree, mHeftVector);
214 if (hi_degree != nullptr)
215 kb_target_hi_weight =
216 exponents::weight(mHeftVector.size(), hi_degree, mHeftVector);
217
218 if (lo_degree && hi_degree && mHeftVector.size() == 1 && mHeftVector[0] < 0)
219 {
220 int t = kb_target_lo_weight;
221 kb_target_lo_weight = kb_target_hi_weight;
222 kb_target_hi_weight = t;
223 }
224
225 kb_mon = M->make_one();
226
227 mat = MatrixConstructor(bottom->rows(), 0);
228 kb_exp_multidegree = D->make_one();
229
230 if (mHeftVector.size() > 1 && lo_degree != nullptr)
231 {
232 kb_target_multidegree = D->make_one();
233 exponents::copy(mHeftVector.size(), lo_degree, kb_target_multidegree);
234 }
235 else { kb_target_multidegree = nullptr; }
236}

249inline bool KBasis::backtrack(int &curr)
250{
251 // if we are the end, decrease the last entry to 0
252 if (curr == mVariables.size() - 1)
253 {
254 kb_exp_weight -= var_wts[curr] * kb_exp[mVariables[curr]];
255 kb_exp[mVariables[curr]] = 0;
256 do {
257 curr--;
258 } while (curr >= 0 && kb_exp[mVariables[curr]] == 0);
259 }
260 if (curr < 0) return false;
261 kb_exp[mVariables[curr]]--;
262 kb_exp_weight -= var_wts[curr];
263 curr++;
264 return true;
265}

276{

323inline bool KBasis::backtrack_mg(int &curr)
324{
325 int vcurr = -1;
326 // if we are the end, decrease the last entry to 0
327 if (curr == mVariables.size() - 1)
328 {
329 vcurr = mVariables[curr];
330 kb_exp_weight -= var_wts[curr] * kb_exp[vcurr];
331 exponents::multpower(mHeftVector.size(),
332 kb_exp_multidegree,
333 var_degs + (mHeftVector.size() * curr),
334 -kb_exp[vcurr],
335 kb_exp_multidegree);
336 kb_exp[vcurr] = 0;
337 do {
338 curr--;
339 } while (curr >= 0 && kb_exp[mVariables[curr]] == 0);
340 }
341 if (curr < 0) return false;
342 vcurr = mVariables[curr];
343 kb_exp[vcurr]--;
344 kb_exp_weight -= var_wts[curr];
345 exponents::divide(mHeftVector.size(),
346 kb_exp_multidegree,
347 var_degs + (mHeftVector.size() * curr),
348 kb_exp_multidegree);
349 curr++;
350 return true;
351}

394static bool all_have_pure_powers(const MonomialIdeal *M,
395 std::vector<int> varlist)
396{
397 // returns true iff all the variables in varlist have some pure power in M
398 std::vector<int> lcms = M2_arrayint_to_stdvector<int>(M->lcm());
399 exponents_t exp = ALLOCATE_EXPONENTS(EXPONENT_BYTE_SIZE(lcms.size()));
400 for (int i = 0; i < lcms.size(); i++) exp[i] = 0;
401 for (int i = 0; i < varlist.size(); i++)
402 {
403 Bag *b;
404 int v = varlist[i];
405 exp[v] = lcms[v];
406 if (!M->search_expvector(exp, b)) return false;
407 exp[v] = 0;
408 }
409 return true;
410}

412void KBasis::compute()
413// Only the lead monomials of the two matrices 'this' and 'bottom' are
414// considered. Thus, you must perform the required GB's elsewhere.
415// Find a basis for (image this)/(image bottom) in degree d.
416// If 'd' is NULL, first check that (image this)/(image bottom) has
417// finite dimension, and if so, return a basis.
418// If 'd' is not NULL, it is an element of the degree monoid.
419{
420 if (limit == 0) return;
421 std::vector<int> zero_vars;
422 if (weight_has_zeros)
423 {
424 for (int i = 0; i < mVariables.size(); i++)
425 {
426 if (var_wts[i] == 0) { zero_vars.push_back(i); }
427 }
428 }
429
430 for (int i = 0; i < bottom_matrix->n_rows(); i++)
431 {
432 if (system_interrupted()) return;
433 kb_comp = i;
434
435 // Make the monomial ideal: this should contain only
436 // monomials involving 'mVariables'.
437 kb_monideal = bottom_matrix->make_monideal(i, true);
438 // the true means: over ZZ, don't consider monomials with non-unit lead
439 // coeffs
440
441 if (kb_monideal->is_one()) continue;
442 if (hi_degree == nullptr)
443 {
444 // check here that kb_monideal is 0-dimensional
445 // (at least for the variables being used):
446 if (!all_have_pure_powers(kb_monideal, mVariables))
447 {
448 kb_error = true;
449 ERROR("module given is not finite over the base");
450 return;
451 }
452 }
453 else if (zero_vars.size() > 0)
454 {
455 // if we have any variables with zero degrees, then kb_monideal needs
456 // to be 0-dimensional in those variables.
457 if (!all_have_pure_powers(kb_monideal, zero_vars))
458 {
459 kb_error = true;
460 ERROR(
461 "module given is not finite over the zero-degree variables");
462 return;
463 }
464 }
465
466 const int *component_degree = bottom_matrix->rows()->degree(i);
467 D->to_expvector(component_degree, kb_exp_multidegree);
468 kb_exp_weight =
469 exponents::weight(mHeftVector.size(), kb_exp_multidegree, mHeftVector);
470
471 // Do the recursion
472 switch (computation_type)
473 {
474 case KB_FULL:
475 basis0_full();
476 break;
477 case KB_SINGLE:
478 basis0_singly_graded();
479 break;
480 case KB_MULTI:
481 basis0_multi_graded();
482 break;
483 }
484 }
485}

487Matrix /* or null */ *KBasis::k_basis(const Matrix *bottom,
488 M2_arrayint lo_degree,
489 M2_arrayint hi_degree,
490 std::vector<int> heftvec,
491 std::vector<int> varlist,
492 bool do_truncation,
493 int limit)
494{
495 // There are essentially 3 situations:
496 // (a) basis(M) -- lo_degree and hi_degree are not given
497 // in this case, only need that for each variable in 'varlist',
498 // some power is an initial term of 'bottom' (for each row of 'bottom').
499 // heft is not used here, or considered.
500 // (b) basis(lo,hi,M) -- case when the ring is singly-graded
501 // one of lo and hi must be given. (otherwise we are in case (a) above)
502 // In this case, heft is a list with one element in it.
503 // Assume: heft * deg(x) > 0, for all x in 'varlist'.
504 // In this situation: we use kb_target_lo_heft, kb_target_hi_heft
505 // (c) basis(d, d, M) -- ring is multi-graded
506 // ASSUME: deg_d(x) . heft > 0 for all vars 'x' in 'varlist'
507 // where deg_d(x) consists of the first #d components of deg(x)
508 // ASSUME: 1 <= #d <= degreeRank of the ring
509 // use kb_target_multidegree, kb_target_lo_heft
510 // and kb_exp_multidegree (of length #d).
511 // if do_truncation, then any generator with heft > kb_target_lo_heft is
512 // placed in the resulting matrix
513 //
514 // Further assumptions:
515 // 1 <= #heft <= degreeRank P
516 // #heft = #lo_degree = #hi_degree, if these are not 0.
517 //
518 //
519 // Do some checks first, return 0 if not good.
520 const PolynomialRing *P = bottom->get_ring()->cast_to_PolynomialRing();
521 if (P == nullptr) return Matrix::identity(bottom->rows());
522
523 const PolynomialRing *D = P->get_degree_ring();
524 const int *lo = lo_degree->len > 0 ? lo_degree->array : nullptr;
525 const int *hi = hi_degree->len > 0 ? hi_degree->array : nullptr;
526
527 if (heftvec.size() > D->n_vars())
528 {
529 ERROR("expected heft vector of length <= %d", D->n_vars());
530 return nullptr;
531 }
532
533 if (lo && heftvec.size() != lo_degree->len)
534 {
535 ERROR("expected degrees of length %d", heftvec.size());
536 return nullptr;
537 }
538
539 if (hi && heftvec.size() != hi_degree->len)
540 {
541 ERROR("expected degrees of length %d", heftvec.size());
542 return nullptr;
543 }
544
545 // If heftvec.size() is > 1, and both lo and hi are non-null,
546 // they need to be the same
547 if (heftvec.size() > 1 && lo && hi)
548 for (int i = 0; i < heftvec.size(); i++)
549 if (lo_degree->array[i] != hi_degree->array[i])
550 {
551 ERROR("expected degree bounds to be equal");
552 return nullptr;
553 }
554
555 KBasis KB(bottom, lo, hi, heftvec, varlist, do_truncation, limit);
556
557 // If either a low degree, or high degree is given, then we require a
558 // non-negative heft vector:
559 if (lo || hi)
560 for (int i = 0; i < varlist.size(); i++)
561 if (KB.var_wts[i] < 0)
562 {
563 ERROR(
564 "basis: computation requires a heft form non-negative on the "
565 "degrees of the variables");
566 return nullptr;
567 }
568 // This next line will happen if both lo,hi degrees are given, and they are
569 // different. This can only be the singly generated case, and in that case
570 // the degrees are in the wrong order, so return with 0 basis.
571 if (lo != nullptr && hi != nullptr && lo[0] > hi[0]) return KB.value();
572
573 KB.compute();
574 if (system_interrupted()) return nullptr;
575 return KB.value();
576}
