59void MarkedGB::add_marked_elems(const VECTOR(gbvector *) & leadterms0,
60 const VECTOR(POLY) & polys0,
61 bool auto_reduced)
62{
63 (void) auto_reduced;
64 // First sort these elements via increasing lex order (or monomial order?)
65 // Next insert minimal elements into T, and polys
66 const Monoid *M = originalR->getMonoid();
67
68 leadterms = newarray(gbvector *, leadterms0.size());
69
70 // Now loop through each element, and see if the lead monomial is in T.
71 // If not, add it in , and place element into 'polys'.
72
73 for (int i = 0; i < leadterms0.size(); i++)
74 {
75 POLY h;
76 ring_elem junk;
77
78 gbvector *f = polys0[i].f;
79 gbvector *inf = leadterms0[i];
80
81 h.f = R->gbvector_copy(f);
82 h.fsyz = R->gbvector_copy(polys0[i].fsyz);
83
84 gbvector *iinf = nullptr;
85 for (gbvector *t = h.f; t != nullptr; t = t->next)
86 if (inf->comp == t->comp && EQ == M->compare(inf->monom, t->monom))
87 {
88 iinf = t;
89 break;
90 }
91 if (!iinf)
92 {
93 ERROR("lead term does not appear in the polynomial!");
94 iinf = f;
95 }
96 leadterms[i] = iinf;
97
98 exponents_t e = R->exponents_make();
99 R->gbvector_get_lead_exponents(F, iinf, e);
100
101 // if (auto_reduced)
102 // remainder(h,false,junk); // This auto-reduces h. MES MES!! Maybe
103 // not for marked GB!!
104
105 R->gbvector_remove_content(h.f, h.fsyz);
106
107 T->insert(e, iinf->comp, i);
108 polys.push_back(h);
109 }
110
111 auto_reduce();
112}

124void MarkedGB::marked_remainder(POLY &f,
125 bool use_denom,
126 ring_elem &denom,
127 gbvector *marked_lead_term)
128// If marked_lead_term is not NULL, then it should be a pointer
129// to a term in f.f. This term will not be reduced, and the new lead term will
130// replace this one.
131// (This could happen if the coefficient changes).
132{
133 gbvector head;
134 gbvector *frem = &head;
135 frem->next = nullptr;
136 POLY h = f;
137 exponents_t EXP = ALLOCATE_EXPONENTS(R->exponent_byte_size());
138
139 while (!R->gbvector_is_zero(h.f))
140 {
141 if (h.f != marked_lead_term)
142 {
143 R->gbvector_get_lead_exponents(F, h.f, EXP);
144 int x = h.f->comp;
145 int w = T->find_divisor(EXP, x);
146 if (w >= 0)
147 {
148 POLY g = polys[w];
149 R->gbvector_reduce_with_marked_lead_term(F,
150 Fsyz,
151 head.next,
152 h.f,
153 h.fsyz,
154 leadterms[w],
155 g.f,
156 g.fsyz,
157 use_denom,
158 denom);
159 continue;
160 }
161 }
162 frem->next = h.f;
163 frem = frem->next;
164 h.f = h.f->next;
165 frem->next = nullptr;
166 }
167
168 h.f = head.next;
169 f.f = h.f;
170 f.fsyz = h.fsyz;
171 R->gbvector_sort(F, f.f);
172 R->gbvector_sort(Fsyz, f.fsyz);
173}

222void MarkedGB::geo_remainder(gbvector *&f, bool use_denom, ring_elem &denom)
223{
224 gbvector head;
225 gbvector *frem = &head;
226 (void) use_denom;
227 (void) denom;
228 frem->next = nullptr;
229
230 gbvectorHeap fb(R, F);
231 gbvectorHeap zero(R, Fsyz);
232 fb.add(f);
233
234 const gbvector *lead;
235 exponents_t EXP = ALLOCATE_EXPONENTS(R->exponent_byte_size());
236 while ((lead = fb.get_lead_term()) != nullptr)
237 {
238 R->gbvector_get_lead_exponents(F, lead, EXP);
239 int x = lead->comp;
240 int w = T->find_divisor(EXP, x);
241 if (w < 0)
242 {
243 frem->next = fb.remove_lead_term();
244 frem = frem->next;
245 frem->next = nullptr;
246 }
247 else
248 {
249 POLY g = polys[w];
250 R->reduce_marked_lead_term_heap(
251 F, Fsyz, lead, EXP, head.next, fb, zero, leadterms[w], g.f, nullptr);
252 }
253 }
254 f = head.next;
255 R->gbvector_sort(F, f);
256}
