950class PriorityQueuePolynomialHeap : public PolynomialHeap
951{
952public:
953 using Entry = std::pair<Monom, ring_elem>;
954 using Container = std::vector<Entry, StatsAllocator<Entry>>;
955
956 PriorityQueuePolynomialHeap(const FreeAlgebra& F)
957 : mRing(F),
958 mEntryConfig(F.monoid()),
959 mQueue(mEntryConfig),
960 mLeadTermSet(false),
961 mLeadTerm(Monom(), F.coefficientRing()->zero())
962 {
963 }
964
965 virtual ~PriorityQueuePolynomialHeap() {}
966
967 void clear() override {
968 // clear the heap. The free algebra is kept the same
969 // but all other aspects are reset. The MonomialSpace
970 // has all its data freed to the arena, but is available for
971 // use without any new allocations for the next computation.
972 mLeadTermSet = false;
973 while (!mQueue.empty())
974 mQueue.pop();
975 mMonomialSpace.deallocateAll();
976 }
977
978 // prevent copy and assignment constructors
979 // allow move constructors, I guess?
980 PriorityQueuePolynomialHeap operator=(const PriorityQueuePolynomialHeap&) = delete;
981 PriorityQueuePolynomialHeap(const PriorityQueuePolynomialHeap&) = delete;
982
983 PriorityQueuePolynomialHeap& addEntry(const Entry& entry)
984 {
985 auto rg = mMonomialSpace.allocateArray<int>(entry.first.size());
986 std::copy(entry.first.begin(), entry.first.end(), rg.first);
987 mQueue.push(Entry(Monom(rg.first), entry.second));
988 return *this;
989 }
990
991 PriorityQueuePolynomialHeap& addPolynomial(const Poly& poly) override
992 {
993 if (mLeadTermSet)
994 {
995 mQueue.push(mLeadTerm);
996 mLeadTermSet = false;
997 }
998 for (auto i = poly.cbegin(); i != poly.cend(); ++i)
999 addEntry(Entry(i.monom(),i.coeff()));
1000 return *this;
1001 }
1002
1003 PriorityQueuePolynomialHeap& addPolynomial(ring_elem coeff,
1004 Word left,
1005 Word right,
1006 const Poly& poly) override
1007 {
1008
1009 Poly f;
1010 mRing.mult_by_term_left_and_right(f, poly, coeff, left, right);
1011 addPolynomial(f);
1012 return *this;
1013 }
1014
1015 bool isZero() override
1016 {
1017 if (mLeadTermSet) return false;
1018 if (mQueue.empty()) return true;
1019 Entry lt = mQueue.top();
1020 mQueue.pop();
1021 while (not mQueue.empty())
1022 {
1023 Entry e = mQueue.top();
1024 if (mRing.monoid().compare(e.first, lt.first) == EQ)
1025 {
1026 lt.second = mRing.coefficientRing()->add(lt.second, e.second);
1027 mQueue.pop();
1028 }
1029 else
1030 {
1031 if (not mRing.coefficientRing()->is_zero(lt.second))
1032 {
1033 mLeadTermSet = true;
1034 mLeadTerm = lt;
1035 return false;
1036 }
1037 else
1038 {
1039 lt = mQueue.top();
1040 mQueue.pop();
1041 }
1042 }
1043 }
1044 if (not mRing.coefficientRing()->is_zero(lt.second))
1045 {
1046 mLeadTermSet = true;
1047 mLeadTerm = lt;
1048 return false;
1049 }
1050 return true;
1051 }
1052
1053 std::pair<Monom, ring_elem> viewLeadTerm() override
1054 {
1055 if (isZero())
1056 {
1057 std::cout << "viewLeadTerm called without checking if polynomial is zero" << std::endl;
1058 assert(false);
1059 }
1060 assert(mLeadTermSet);
1061 return mLeadTerm;
1062 }
1063
1064 void removeLeadTerm() override
1065 {
1066 if (isZero()) return;
1067 assert(mLeadTermSet); // should only be called if mLeadTermSet is true.
1068 if (mLeadTermSet)
1069 mLeadTermSet = false;
1070 }
1071
1072 Poly* value() override
1073 {
1074 Poly* f = new Poly;
1075 if (mLeadTermSet)
1076 mQueue.push(mLeadTerm);
1077 mLeadTermSet = false;
1078 while (not isZero())
1079 {
1080 auto tm = viewLeadTerm();
1081 mRing.add_to_end(*f, tm.second, tm.first);
1082 removeLeadTerm();
1083 }
1084 addPolynomial(*f);
1085 return f;
1086 }
1087
1088 size_t getMemoryUsedInBytes() override
1089 {
1090 return (mQueue.size()*sizeof(Entry)) + mMonomialSpace.getMemoryUsedInBytes();
1091 }
1092
1093 std::string getName() const override { return "Priority queue heap."; }
1094
1095private:
1096 FreeAlgebra mRing;
1097 EntryConfig mEntryConfig;
1098 std::priority_queue<Entry,Container,EntryConfig> mQueue;
1099 MemoryBlock mMonomialSpace;
1100 bool mLeadTermSet; // true means mLeadTerm is set, to a non-zero value.
1101 Entry mLeadTerm;
1102};
