187 bool from_expvector(const_ntuple_monomial e,
188 long comp,
189 packed_monomial result) const
190 {
191 // Pack the vector e[0]..e[nvars-1],comp. Create the hash value at the same
192 // time.
193 ncalls_from_expvector++;
194 result[0] = 0;
195 result[1] = comp;
196
197 for (int i = 0; i < nvars; i++)
198 {
199 result[firstvar + i] = e[i];
200 if (e[i] > 0) result[0] += hashfcn[i] * e[i];
201 }
202
203 const int *wt = mWeightVectors.data();
204 for (int j = 0; j < mNumWeights; j++, wt += nvars)
205 {
206 long val = 0;
207 for (int i = 0; i < nvars; i++)
208 {
209 long a = e[i];
210 if (a > 0) val += a * wt[i];
211 }
212 result[2 + j] = val;
213 }
214 return true;
215 }

289 long comp,

291 {

376 bool divide(const_packed_monomial m,
377 const_packed_monomial n,
378 packed_monomial result) const
379 {
380 ncalls_divide++;
381 // First, divide monomials
382 // Then, if the division is OK, set the component, hash value and rest of
383 // the monomial
384 if (m[1] != n[1]) // components are not equal
385 return false;
386 const_packed_monomial m1 = m + nslots;
387 const_packed_monomial n1 = n + nslots;
388 packed_monomial result1 = result + nslots;
389 for (int i = nslots - 2; i > 0; i--)
390 {
391 varpower_word cmp = *--m1 - *--n1;
392 if (cmp < 0) return false;
393 *--result1 = cmp;
394 }
395 result[1] = 0; // the component of a division is in the ring (comp 0).
396 result[0] = m[0] - n[0]; // subtract hash codes
397 return true;
398 }

512 // returns LT, EQ, GT:
513 // LT if m < n in the monomial order
514 // GT if m > n
515 // otherwise EQ
516 int compare(const_packed_monomial m,
517 const_packed_monomial n) const
518 {
519 ncalls_compare++;
520 const_packed_monomial m1 = m + 2;
521 const_packed_monomial n1 = n + 2;
522 for (int i = 0; i < mNumWeights; i++)
523 {
524 varpower_word cmp = *m1++ - *n1++;
525 if (cmp > 0) return GT;
526 if (cmp < 0) return LT;
527 }
528 if (mTieBreakerIsRevLex)
529 {
530 m1 = m + nslots;
531 n1 = n + nslots;
532 for (int i = nvars - 1; i >= 0; i--)
533 {
534 varpower_word cmp = *--m1 - *--n1;
535 if (cmp < 0) return GT;
536 if (cmp > 0) return LT;
537 }
538 } else {
539 m1 = m + firstvar;
540 n1 = n + firstvar;
541 for (int i = 0; i < nvars; ++i)
542 {
543 varpower_word cmp = *m1++ - *n1++;
544 if (cmp > 0) return GT;
545 if (cmp < 0) return LT;
546 }
547 }
548
549 monomial_word cmp = m[1] - n[1];
550 if (cmp < 0) return LT;
551 if (cmp > 0) return GT;
552 return EQ;
553 }

570 bool unnecessary1(const_packed_monomial m,
571 const_packed_monomial p1,
572 const_packed_monomial p2,
573 const_packed_monomial lcm) const
574 // Returns true if the corresponding pair (p1,p2) could be removed
575 // This is essentially the Buchberger-Moeller criterion
576 // Assumptions: lcm(p1,p2) = lcm.
577 // Here is the criterion:
578 // (a) if component(lcm) != component(m) return false
579 // (b) if m does not divide lcm, return false
580 // (c) need that (A) lcm(p1,m) != lcm and that (B) lcm(p2,m) != lcm
581 // (in these two cases, we will have already removed one of the other
582 // two pairs: (p1,m), (p2,m). Note that in any case,
583 // if (b) holds, then lcm(p1,m) divides lcm, same with lcm(p2,m).
584 // if A and B then return true
585 {
586 // TODO: MES: this maybe should check that lead components of m and p1 and
587 // p2 are the same...
588 ncalls_unneccesary++;
589 bool A = false;
590 bool B = false;
591 m += firstvar;
592 p1 += firstvar;
593 p2 += firstvar;
594 lcm += firstvar;
595 for (int i = 0; i < nvars; i++)
596 {
597 if (m[i] > lcm[i]) return false;
598 if (m[i] == lcm[i]) continue;
599 if (!A && p1[i] < lcm[i])
600 {
601 A = true;
602 continue;
603 }
604 if (!B && p2[i] < lcm[i])
605 {
606 B = true;
607 }
608 }
609 return (A && B);
610 }

612 bool unnecessary(const_packed_monomial m,
613 const_packed_monomial p1,
614 const_packed_monomial p2,
615 const_packed_monomial lcm) const
616 // Returns true if the corresponding pair (p1,p2) could be removed
617 // This is essentially the Buchberger-Moeller criterion
618 // Assumptions: lcm(p1,p2) = lcm.
619 // Here is the criterion:
620 // (a) if component(lcm) != component(m) return false
621 // (b) if m does not divide lcm, return false
622 // (c) need that (A) lcm(p1,m) != lcm and that (B) lcm(p2,m) != lcm
623 // (in these two cases, we will have already removed one of the other
624 // two pairs: (p1,m), (p2,m). Note that in any case,
625 // if (b) holds, then lcm(p1,m) divides lcm, same with lcm(p2,m).
626 // if A and B then return true
627
628 // Here is the criterion to remove a pair (i.e. return true)
629 // (a) need: component(lcm) == component(m) (else return false)
630 // (b) need: m divides lcm, (else return false)
631 // (c) need: (A) and (B), where
632 // (A) lcm(p1,m) != lcm
633 // (B) lcm(p2,m) != lcm
634 //
635 // (in these two cases, we will have already removed one of the other
636 // two pairs: (p1,m), (p2,m). Note that in any case,
637 // if (b) holds, then lcm(p1,m) divides lcm, same with lcm(p2,m).
638 // if A and B then return true
639
640 {
641 // TODO: MES: this maybe should check that lead components of m and p1 and
642 // p2 are the same...
643 ncalls_unneccesary++;
644 bool A = false;
645 m += firstvar;
646 p1 += firstvar;
647 p2 += firstvar;
648 lcm += firstvar;
649 for (int i = 0; i < nvars; i++)
650 if (m[i] > lcm[i]) return false;
651
652 for (int i = 0; i < nvars; i++)
653 {
654 varpower_word a = lcm[i];
655 if (p1[i] < a && m[i] < a)
656 {
657 A = true;
658 break;
659 }
660 }
661
662 if (!A) return false;
663
664 // Now (b), (A) hold. Check (B).
665 for (int i = 0; i < nvars; i++)
666 {
667 varpower_word a = lcm[i];
668 if (p2[i] < a && m[i] < a) return true;
669 }
670
671 return false;
672 }

681 void quotient_as_vp(const_packed_monomial a,
682 const_packed_monomial b,
683 varpower_monomial result,
684 int &deg_result,
685 bool &are_disjoint) const
686 {
687 varpower_word deg = mModuleHeftDegrees[get_component(a)];
688 // sets result, deg, are_disjoint
689 ncalls_quotient_as_vp++;
690 are_disjoint = true;
691 a += firstvar;
692 b += firstvar;
693 int len = 0;
694 varpower_word *r = result + 1;
695 for (int i = nvars - 1; i >= 0; --i)
696 {
697 if (a[i] != 0 && b[i] != 0) are_disjoint = false;
698 long c = a[i] - b[i];
699 if (c > 0)
700 {
701 *r++ = i;
702 *r++ = c;
703 deg += c * mHeftDegrees[i];
704 len++;
705 }
706 }
707 result[0] = len;
708 deg_result = static_cast<int>(deg);
709 }
