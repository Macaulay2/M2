43bool almostEqual(const M2::ARingCCC& C,
44 int nbits,
45 const M2::ARingCCC::ElementType& a,
46 const M2::ARingCCC::ElementType& b)
47{
48 mpfr_t epsilon;
49 mpfr_init2(epsilon, C.get_precision());
50 mpfr_set_ui_2exp(epsilon, 1, -nbits, MPFR_RNDN); // should there be exp() ???
51
52 M2::ARingCCC::ElementType c;
53 C.init(c);
54 C.subtract(c, a, b);
55
56 bool ret = mpfr_cmpabs(&c.re, epsilon) < 0 && mpfr_cmpabs(&c.im, epsilon) < 0;
57
58 C.clear(c);
59 mpfr_clear(epsilon);
60 return ret;
61}

190TEST(ARingCCC, axioms)
191{
192 M2::ARingCCC C(100);
193 ARingElementGenerator<M2::ARingCCC> gen(C);
194 M2::ARingCCC::ElementType a, b, c, d, e;
195 C.init(a);
196 C.init(b);
197 C.init(c);
198 C.init(d);
199 C.init(e);
200 for (int i = 0; i < ntrials; i++)
201 {
202 gen.nextElement(a);
203 gen.nextElement(b);
204 gen.nextElement(c);
205 // Test commutativity
206 // test: a*b = b*a
207 // test: a+b == b+a
208 C.add(d, a, b);
209 C.add(e, b, a);
210 EXPECT_TRUE(almostEqual(C, 98, d, e));
211 C.mult(d, a, b);
212 C.mult(e, b, a);
213 EXPECT_TRUE(almostEqual(C, 98, d, e));
214
215 // Test associativity
216 // test: a+(b+c) == (a+b)+c
217 // test: a*(b*c) == (a*b)*c
218 C.add(e, b, c);
219 C.add(d, a, e); // a+(b+c)
220 C.add(e, a, b);
221 C.add(e, e, c); // (a+b)+c
222 EXPECT_TRUE(almostEqual(C, 94, d, e));
223 C.mult(e, b, c);
224 C.mult(d, a, e); // a*(b*c)
225 C.mult(e, a, b);
226 C.mult(e, e, c); // (a*b)*c
227 EXPECT_TRUE(almostEqual(
228 C, 93, d, e)); // MES: I'm not sure how equal these should be.
229
230 // Test distributivity
231 // test: a*(b+c) == a*b + a*c
232 C.add(e, b, c);
233 C.mult(d, a, e); // a*(b+c)
234 C.mult(b, a, b);
235 C.mult(c, a, c);
236 C.add(e, b, c); // a*b + a*c
237 EXPECT_TRUE(almostEqual(C, 93, d, e));
238 }
239 C.clear(e);
240 C.clear(d);
241 C.clear(c);
242 C.clear(b);
243 C.clear(a);
244}
