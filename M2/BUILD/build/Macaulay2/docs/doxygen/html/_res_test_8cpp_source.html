125TEST(ResMonoidDense, mult)
126{
127 ResMonoidDense M(4,
128 std::vector<int>{1, 1, 1, 1},
129 std::vector<int>{},
130 MonomialOrderingType::GRevLex);
131
132 int exp1[]{0, 0, 0, 0};
133 int exp2[]{0, 1, 0, 3};
134 int exp3[]{0, 0, 0, 0};
135 int mon1[]{0, 0, 0, 0, 0, 0};
136 int mon2[]{0, 0, 0, 0, 0, 0, 0, 0, 0};
137 int mon3[]{0, 0, 0, 0, 0, 0, 0, 0, 0};
138 int mon[]{0, 0, 0, 0, 0, 0, 0, 0, 0};
139 int comp;
140 M.from_expvector(exp1, 0, mon1);
141 M.from_expvector(exp2, 0, mon2);
142 M.mult(mon1, mon2, mon);
143 M.to_expvector(mon, exp3, comp);
144 EXPECT_EQ(6, M.monomial_size(mon1));
145 EXPECT_EQ(6, M.monomial_size(mon2));
146 EXPECT_EQ(6, M.monomial_size(mon));
147
148 for (int i = 0; i < M.n_vars(); i++)
149 for (int j = i; j < M.n_vars(); j++)
150 for (int k = 0; k < M.n_vars(); k++)
151 {
152 int exp1[]{0, 0, 0, 0};
153 exp1[i]++;
154 exp1[j]++;
155 int exp2[]{0, 0, 0, 0};
156 exp2[k]++;
157 M.from_expvector(exp1, 0, mon1);
158 M.from_expvector(exp2, 0, mon2);
159 M.mult(mon1, mon2, mon);
160 M.mult(mon1, mon2, mon3);
161 EXPECT_TRUE(M.is_equal(mon, mon3));
162 M.to_expvector(mon, exp3, comp);
163 int exp3a[]{0, 0, 0, 0};
164 exp3a[i]++;
165 exp3a[j]++;
166 exp3a[k]++;
167 for (int ell = 0; ell < M.n_vars(); ell++)
168 EXPECT_EQ(exp3[ell], exp3a[ell]);
169 EXPECT_EQ(M.monomial_size(mon), M.monomial_size(mon1));
170 EXPECT_EQ(M.monomial_size(mon), M.monomial_size(mon2));
171 }
172}
