287 void power(elem &result, elem a, int n) const
288 {
289 if (a != 0)
290 {
291 result = (a * n) % p1;
292 if (result <= 0) result += p1;
293 }
294 else
295 {
296 // a == 0
297 if (n == 0) result = p1; // the element 1 in this ring.
298 else if (n > 0) result = 0;
