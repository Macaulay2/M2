318 void invert(ElementType &result, const ElementType &a) const
319 // we silently assume that a != 0. If it is, result is set to a^0, i.e. 1
320 {
321 mpfi_t norm, temp;
322 mpfi_init2(norm,get_precision());
323 mpfi_init2(temp,get_precision());
324 mpfi_set(norm,&a.re);
325 mpfi_mul(norm,norm,norm);
326 mpfi_set(temp,&a.im);
327 mpfi_mul(temp,temp,temp);
328 mpfi_add(norm,norm,temp);
329 mpfi_set(&result.re,&a.re);
330 mpfi_mul_si(&result.im,&a.im,-1);
331 mpfi_div(&result.re,&result.re,norm);
332 mpfi_div(&result.im,&result.im,norm);
333 }

375 void mult(ElementType &result,
376 const ElementType &a,
377 const ElementType &b) const
378 {
379 mpfi_t temp, retemp, imtemp;
380 mpfi_init2(temp,get_precision());
381 mpfi_init2(retemp,get_precision());
382 mpfi_init2(imtemp,get_precision());
383 mpfi_mul(retemp,&a.re,&b.re);
384 mpfi_mul(temp,&a.im,&b.im);
385 mpfi_sub(retemp,retemp,temp);
386 mpfi_mul(temp,&a.re,&b.im);
387 mpfi_mul(imtemp,&a.im,&b.re);
388 mpfi_add(&result.im,imtemp,temp);
389 mpfi_set(&result.re,retemp);
390 }

404 void power(ElementType &result, const ElementType &a, int n) const
405 {
406 if (n >= 2)
407 {
408 ElementType b;
409 init(b);
410 if (n%2 == 0)
411 {
412 power(b,a,n/2);
413 mult(result,b,b);
414 }
415 else
416 {
417 power(b,a,n-1);
418 mult(result,a,b);
419 }
420 }
421 else if (n == 1)
422 {
423 mpfi_set(&result.re,&a.re);
424 mpfi_set(&result.im,&a.im);
425 }
426 else if (n == 0)
427 {
428 mpfi_set_si(&result.re,1);
429 mpfi_set_si(&result.im,0);
430 }
431 else if (n<0)
432 throw 20;
433 }
