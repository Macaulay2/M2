40template <typename Mat>
41void displayMat(buffer& o, const Mat& A)
42{
43 // Assumption: Mat is either DMat<RingType> or SMat<RingType>, in that it
44 // defines the following:
45 // Mat::ElementType
46 // A.ring()
47 // A.numRows(), A.numColumns()
48 // A.entry(r,c)
49 // and A.ring().elem_text_out
50
51 size_t nrows = A.numRows();
52 size_t ncols = A.numColumns();
53 buffer* p = new buffer[nrows];
54 size_t r;
55 for (size_t c = 0; c < ncols; c++)
56 {
57 size_t maxcount = 0;
58 for (r = 0; r < nrows; r++)
59 {
60 const typename Mat::ElementType& a = A.entry(r, c);
61 if (!A.ring().is_zero(a))
62 A.ring().elem_text_out(p[r], a, true, false, false);
63 else
64 p[r] << ".";
65 if (p[r].size() > maxcount) maxcount = p[r].size();
66 }
67 for (r = 0; r < nrows; r++)
68 for (size_t k = maxcount + 1 - p[r].size(); k > 0; k--) p[r] << ' ';
69 }
70 for (r = 0; r < nrows; r++)
71 {
72 p[r] << '\0';
73 char* s = p[r].str();
74 o << s << newline;
75 }
76 delete[] p;
77}

87template <typename Mat>
88static void concatenateMatrices(const Mat& A, const Mat& B, Mat& C)
89{
90 assert(A.numRows() == B.numRows());
91 C.resize(A.numRows(), A.numColumns() + B.numColumns());
92 for (long r = 0; r < A.numRows(); r++)
93 for (long c = 0; c < A.numColumns(); c++)
94 A.ring().set(C.entry(r, c), A.entry(r, c));
95 for (long r = 0; r < A.numRows(); r++)
96 for (long c = 0; c < B.numColumns(); c++)
97 A.ring().set(C.entry(r, c + A.numColumns()), B.entry(r, c));
98}
