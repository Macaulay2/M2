193 unsigned int computeHashValue(const elem& a) const
194 {
195 unsigned int hash = 0;
196 long deg = nmod_poly_degree(&a);
197 for (long i = 0; i <= deg; i++)
198 hash += static_cast<unsigned int>(nmod_poly_get_coeff_ui(&a, i)) *
199 mPPowers[i];
200 // printf("computing hash value: %ld\n", hash);

204 void to_ring_elem(ring_elem& result, const ElementType& a) const
205 {
206 ElementType* b = getmemstructtype(ElementType*);
207 init(*b);
208 copy(*b, a);
209 size_t coeffs_size = sizeof(mp_limb_t)*b->alloc;

365 void power(ElementType& result, const ElementType& a, int n) const
366 {
367 if (n < 0)
368 {
369 invert(result, a);
370 fq_nmod_pow_ui(&result, &result, -n, mContext);
371 }
372 else
373 fq_nmod_pow_ui(&result, &a, n, mContext);

376 void power_mpz(ElementType& result, const ElementType& a, mpz_srcptr n) const
377 {
378 if (mpz_sgn(n) < 0 and is_zero(a))
379 throw exc::division_by_zero_error();
380
381 ElementType base;
382 init(base);
383 if (mpz_sgn(n) < 0)
384 invert(base, a);
385 else
386 copy(base, a);
387
388 mpz_t abs_n;
389 mpz_init(abs_n);
390 mpz_abs(abs_n, n);
391
392 fmpz_t fn;
393 fmpz_init_set_readonly(fn, abs_n);
394 fq_nmod_pow(&result, &base, fn, mContext);
395 fmpz_clear_readonly(fn);
396 mpz_clear(abs_n);
397 clear(base);
398 }
