266 void PackedKDTree<C>::insert(const ExtEntry& extEntry) {
267 MATHIC_ASSERT(debugIsValid());
268 // find node in which to insert extEntry
269 typename Node::Child* parentChild = 0;
270 if (_root == 0)
271 _root = Node::makeNode(_arena, _conf);
272 Node* node = _root;
273 typename Node::iterator child = node->childBegin();
274 while (true) {
275 if (child == node->childEnd()) {

287 child->updateToLowerBound(extEntry);
288 if (node->inChild(child, extEntry.get(), _conf)) {
289 parentChild = &*child;
290 node = child->node;
291 child = node->childBegin();
292 } else
293 ++child;
294 }

299 template<class Iter>
300 void PackedKDTree<C>::reset(Iter insertBegin, Iter insertEnd, const DivMaskCalculator& calc) {
301 clear();
302 if (insertBegin == insertEnd)
303 return;
304
305 typedef InsertTodo<Iter> Task;
306 typedef std::vector<Task> TaskCont;
307 TaskCont todo;
308 TaskCont children;
309
310 {
311 Task initialTask;
312 initialTask.begin = insertBegin;
313 initialTask.end = insertEnd;
314 initialTask.var = static_cast<size_t>(-1);
315 initialTask.fromParent = 0;
316 todo.push_back(initialTask);
317 }
318 while (!todo.empty()) {
319 Iter begin = todo.back().begin;
320 Iter end = todo.back().end;
321 size_t var = todo.back().var;
322 typename Node::Child* fromParent = todo.back().fromParent;
323 if (fromParent != 0) {
324 fromParent->var = var;
325 fromParent->exponent = todo.back().exp;
326 }
327 todo.pop_back();
328
329 // split off children until reaching few enough entries
330 while (C::LeafSize < static_cast<size_t>(std::distance(begin, end))) {
331 Task child;
332 Iter middle = KDEntryArray<C, ExtEntry>::
333 split(begin, end, var, child.exp, _conf);
334 MATHIC_ASSERT(begin < middle && middle < end);
335 MATHIC_ASSERT(var < _conf.getVarCount());
336 child.begin = middle;
337 child.end = end;
338 child.var = var;
339 children.push_back(child);
340 // now operate on the equal-or-less part of the range
341 end = middle;
342 }
343 Node* node = Node::makeNode
344 (begin, end, _arena, calc, _conf, children.size());
345 if (_root == 0)
346 _root = node;
347 if (fromParent != 0)
348 fromParent->node = node;
349 for (size_t child = 0; child < children.size(); ++child) {
350 children[child].fromParent = &*(node->childBegin() + child);
351 todo.push_back(children[child]);
352 }
353 children.clear();
354 }
355 MATHIC_ASSERT(_root != 0);
356
357 if (C::UseTreeDivMask) {
358 // record nodes in tree using breadth first search
359 typedef std::vector<Node*> NodeCont;
360 NodeCont nodes;
361 nodes.push_back(_root);
362 for (size_t i = 0; i < nodes.size(); ++i) {
363 Node* node = nodes[i];
364 for (typename Node::iterator child = node->childBegin();
365 child != node->childEnd(); ++child)
366 nodes.push_back(child->node);
367 }
368 // compute div masks in reverse order of breath first search
369 typename NodeCont::reverse_iterator it = nodes.rbegin();
370 typename NodeCont::reverse_iterator end = nodes.rend();
371 for (; it != end; ++it) {
372 Node* node = *it;
373 typedef std::reverse_iterator<typename Node::iterator> riter;
374 riter rbegin = riter(node->childEnd());
375 riter rend = riter(node->childBegin());
376 for (riter child = rbegin; child != rend; ++child) {
377 child->resetDivMask();
378 if (child == rbegin)
379 child->updateToLowerBound(node->entries());
380 else {
381 riter prev = child;
382 --prev;
383 child->updateToLowerBound(*prev);
384 }
385 if (child->node->hasChildren())
386 child->updateToLowerBound(*child->node->childBegin());
387 else
388 child->updateToLowerBound(child->node->entries());
389 }
390 MATHIC_ASSERT(node->debugIsValid());
391 }
392 }
393 MATHIC_ASSERT(debugIsValid());
394 }

396 template<class C>
397 typename PackedKDTree<C>::Entry* PackedKDTree<C>::findDivisor
398 (const ExtMonoRef& extMonomial) {
399 MATHIC_ASSERT(_tmp.empty());
400 if (_root == 0)
401 return 0;
402 Node* node = _root;
403 while (true) {
404 // record relevant children for later processing
405 for (typename Node::const_iterator it = node->childBegin();
406 it != node->childEnd(); ++it) {
407 if (C::UseTreeDivMask &&
408 !it->getDivMask().canDivide(extMonomial.getDivMask()))
409 goto next;
410 if (node->inChild(it, extMonomial.get(), _conf))
411 _tmp.push_back(it->node);
412 }
413
414 // look for divisor in entries of node
415 {
416 typename KDEntryArray<C, ExtEntry>::iterator it =
417 node->entries().findDivisor(extMonomial, _conf);
418 if (it != node->entries().end()) {
419 MATHIC_ASSERT(_conf.divides(it->get(), extMonomial.get()));
420 _tmp.clear();
421 return &it->get();
422 }
423 }
424
425next:
426 // grab next node to process
427 if (_tmp.empty())
428 break;
429 node = _tmp.back();
430 _tmp.pop_back();
431 }
432 MATHIC_ASSERT(_tmp.empty());
433 return 0;
434 }

576 template<class C>
577 bool PackedKDTree<C>::debugIsValid() const {
578#ifndef MATHIC_DEBUG
579 return true;
580#else
581 //print(std::cerr); std::cerr << std::flush;
582 MATHIC_ASSERT(_tmp.empty());
583 MATHIC_ASSERT(!_conf.getDoAutomaticRebuilds() || _conf.getRebuildRatio() > 0);
584 if (_root == 0)
585 return true;
586
587 // record all nodes
588 std::vector<Node*> nodes;
589 nodes.push_back(_root);
590 for (size_t i = 0; i < nodes.size(); ++i) {
591 Node* node = nodes[i];
592 MATHIC_ASSERT(node->entries().debugIsValid());
593 for (typename Node::iterator it = node->childBegin();
594 it != node->childEnd(); ++it) {
595 MATHIC_ASSERT(it->var < _conf.getVarCount());
596 MATHIC_ASSERT(!C::UseTreeDivMask ||
597 it->canDivide(node->entries()));
598 nodes.push_back(it->node);
599 }
600 }
601
602 // check the recorded nodes
603 MATHIC_ASSERT(_tmp.empty());
604 for (size_t i = 0; i < nodes.size(); ++i) {
605 Node* ancestor = nodes[i];
606
607 for (typename Node::iterator ancestorIt = ancestor->childBegin();
608 ancestorIt != ancestor->childEnd(); ++ancestorIt) {
609 MATHIC_ASSERT(_tmp.empty());
610 size_t var = ancestorIt->var;
611 Exponent exp = ancestorIt->exponent;
612 // check strictly greater than subtree
613 _tmp.push_back(ancestorIt->node);
614 while (!_tmp.empty()) {
615 Node* node = _tmp.back();
616 _tmp.pop_back();
617 for (typename Node::iterator it = node->childBegin();
618 it != node->childEnd(); ++it) {
619 MATHIC_ASSERT(!C::UseTreeDivMask || ancestorIt->canDivide(*it));
620 _tmp.push_back(it->node);
621 }
622 MATHIC_ASSERT(node->entries().
623 allStrictlyGreaterThan(var, exp, _conf));
624 MATHIC_ASSERT(!C::UseTreeDivMask ||
625 ancestorIt->canDivide(node->entries()));
626 }
627 // check less than or equal to sub tree.
628 MATHIC_ASSERT(ancestor->entries().
629 allLessThanOrEqualTo(var, exp, _conf));
630 // go through the rest of the children
631 typename Node::iterator restIt = ancestorIt;
632 for (++restIt; restIt != ancestor->childEnd(); ++restIt) {
633 MATHIC_ASSERT(!C::UseTreeDivMask || ancestorIt->canDivide(*restIt));
634 _tmp.push_back(restIt->node);
635 }
636 while (!_tmp.empty()) {
637 Node* node = _tmp.back();
638 _tmp.pop_back();
639 for (typename Node::iterator it = node->childBegin();
640 it != node->childEnd(); ++it) {
641 MATHIC_ASSERT(!C::UseTreeDivMask || ancestorIt->canDivide(*it));
642 _tmp.push_back(it->node);
643 }
644 MATHIC_ASSERT(node->entries().
645 allLessThanOrEqualTo(var, exp, _conf));
646 MATHIC_ASSERT(!C::UseTreeDivMask ||
647 ancestorIt->canDivide(node->entries()));
648 }
649 }
650 }
651 return true;
652#endif
653 }

671 template<class C>
672 typename PackedKDTree<C>::Node* PackedKDTree<C>::Node::splitInsert(
673 const ExtEntry& extEntry,
674 Child* childFromParent,
675 memt::Arena& arena,
676 const C& conf
677 ) {
678 MATHIC_ASSERT(conf.getVarCount() > 0);
679 MATHIC_ASSERT(entries().size() > 0);
680 size_t var;
681 if (hasChildren())
682 var = (childEnd() - 1)->var;
683 else if (childFromParent != 0)
684 var = childFromParent->var;
685 else
686 var = static_cast<size_t>(-1);
687 typename C::Exponent exp;
688
689 // there is not room to add another child, so make a
690 // new Node with more space and put the Node we are splitting
691 // off into the vacated space. It will fit as it has no
692 // children at all.
693
694 typename KDEntryArray<C, ExtEntry>::iterator middle =
695 KDEntryArray<C, ExtEntry>::split
696 (entries().begin(), entries().end(), var, exp, conf, &extEntry);
697
698 // ** copy relevant part of *this into new space
699 Node* copied = makeNode(entries().begin(), middle, arena, conf,
700 std::distance(childBegin(), childEnd()) + 1);
701 std::copy(childBegin(), childEnd(), copied->childBegin());
702 Child& newChild = *(copied->childEnd() - 1);
703 newChild.var = var;
704 newChild.exponent = exp;
705 newChild.node = this;
706 if (childFromParent != 0)
707 childFromParent->node = copied;
708 if (C::UseTreeDivMask) {
709 newChild.resetDivMask();
710 newChild.updateToLowerBound(entries());
711 newChild.updateToLowerBound(extEntry);
712 }
713
714 // ** fix up remaining part of *this to be the child of copied
715 // OK to call std::copy as begin is not in the range [middle, end).
716 std::copy(middle, entries().end(), entries().begin());
717 const size_t targetSize = std::distance(middle, entries().end());
718 while (entries().size() > targetSize)
719 entries().pop_back();
720 if (conf.getSortOnInsert())
721 std::sort(entries().begin(), entries().end(), Comparer<C>(conf));
722 if (C::UseTreeDivMask)
723 entries().recalculateTreeDivMask();
724 _childrenEnd = childBegin();
725
726 if (copied->inChild(copied->childEnd() - 1, extEntry.get(), conf))
727 entries().insert(extEntry, conf);
728 else
729 copied->entries().insert(extEntry, conf);
730
731 MATHIC_ASSERT(debugIsValid());
732 MATHIC_ASSERT(copied->debugIsValid());
733
734 return copied;
735 }

740 return true;
741#else
742 MATHIC_ASSERT(entries().debugIsValid());
743 MATHIC_ASSERT(childBegin() <= childEnd());
744 if (C::UseTreeDivMask) {
