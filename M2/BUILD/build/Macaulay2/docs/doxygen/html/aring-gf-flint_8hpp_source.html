358 void power_mpz(ElementType& result, const ElementType& a, mpz_srcptr n) const
359 {
360 if (mpz_sgn(n) < 0)
361 invert(result, a);
362 else
363 copy(result, a);
364
365 mpz_t abs_n;
366 mpz_init(abs_n);
367 mpz_abs(abs_n, n);
368
369 fmpz_t fn;
370 fmpz_init_set_readonly(fn, abs_n);
371 fq_zech_pow(&result, &result, fn, mContext);
372 fmpz_clear_readonly(fn);
373 mpz_clear(abs_n);
374 }
