61 typedef int gbelem_type;
62 // ELEM_IN_RING: a bit set. If this is set, then the next two bits need to
63 // off.
64 // ELEM_MINGEN: a bit set: 0 means it is provably not needed to
65 // generated the ideal or submodule. 1 means it might be a minimal
66 // generator.
67 // NOTE: often the min gens obtained this way are not so good. Exceptions
68 // are
69 // for graded submodules. Then the meaning of 1 is: it IS a min gen.
70 // ELEM_MINGB: a bit set: 0 means is not part of the min gb (so far). 1 means
71 // it is.
72 // SO: possible values are 0, 1, 2, 4, 6.
73 // Test using code like this:
74 // if (g->minlevel & ELEM_MINGEN) { .. do this if it is a possible min gen
75 // .. } else { .. if not .. };

85 // change this value...

87 gbelem_type minlevel;
